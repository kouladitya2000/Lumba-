<#
.SYNOPSIS
  Sanity check for the report created by 1-Generate-SubscriptionTagsReport.ps1.

  Re-reads the live Azure tags for every subscription in the report and confirms that
  BA ID, EIM and EHC ID in the report are EXACTLY what is on the live subscription.
  (It also confirms the subscription name in the report equals the live name.)

  READ-ONLY and FILE-FREE:
    - Report workbook is opened read-only (nothing saved or edited).
    - Azure: only reads (Get-AzSubscription, Get-AzTag). Nothing is changed.
    - Writes NOTHING to disk. Logs and the final summary print on screen.

.PREREQUISITES
  - Microsoft Excel installed (to read the report; no extra modules).
  - Install-Module Az.Accounts, Az.Resources -Scope CurrentUser
  - Connect-AzAccount

.USAGE
  1. Set $ReportExcelPath below to the report created by script 1 (or pass -ReportExcelPath).
  2. Run the script. The report can be open or closed in Excel.
#>

[CmdletBinding()]
param(
    # >>> SET THE PATH OF THE REPORT CREATED BY SCRIPT 1 <<<
    [string]$ReportExcelPath = "C:\Path\To\SubscriptionTagsReport.xlsx",

    # Leave blank to use the first sheet (script 1 names it "SubscriptionTags")
    [string]$SheetName       = "",

    [int]$MaxRetries         = 3
)

$ErrorActionPreference = "Stop"

# Report column header -> Azure tag name
$TagNames = @("BA ID", "EIM", "EHC ID")

function Write-Log {
    param([string]$Message, [string]$Color = "")
    $line = "[{0}] {1}" -f (Get-Date -Format "HH:mm:ss"), $Message
    if ($Color) { Write-Host $line -ForegroundColor $Color } else { Write-Host $line }
}
function Write-Plain {
    param([string]$Message = "", [string]$Color = "")
    if ($Color) { Write-Host $Message -ForegroundColor $Color } else { Write-Host $Message }
}

# ---------- Pre-flight ----------
if (-not (Get-Module -ListAvailable -Name Az.Resources)) {
    throw "Az.Resources module not found. Run: Install-Module Az.Accounts, Az.Resources -Scope CurrentUser"
}
if (-not (Get-AzContext)) { throw "Not signed in to Azure. Run Connect-AzAccount first." }
if (-not (Test-Path -LiteralPath $ReportExcelPath)) {
    throw "Report file not found: $ReportExcelPath  (set the correct path at the top of the script)"
}
$fullReport = (Resolve-Path -LiteralPath $ReportExcelPath).Path

# ---------- Excel helpers (read-only) ----------
function ConvertTo-CellString($v) {
    if ($null -eq $v) { return "" }
    if ($v -is [double]) { return $v.ToString("0.##########", [Globalization.CultureInfo]::InvariantCulture) }
    return [string]$v
}

function Get-SheetData {
    param([string]$FullPath, [string]$Sheet)

    $excel = $null; $wb = $null; $ws = $null; $own = $false; $running = $null; $data = $null

    try { $running = [Runtime.InteropServices.Marshal]::GetActiveObject("Excel.Application") } catch { $running = $null }
    if ($running) {
        foreach ($w in $running.Workbooks) {
            if ($w.FullName -ieq $FullPath) { $wb = $w; $excel = $running; break }
        }
    }

    try {
        if (-not $wb) {
            $excel = New-Object -ComObject Excel.Application
            $excel.Visible = $false
            $excel.DisplayAlerts = $false
            $own = $true
            $wb = $excel.Workbooks.Open($FullPath, 0, $true)      # UpdateLinks=0, ReadOnly=true
        }
        $ws   = if ($Sheet) { $wb.Worksheets.Item($Sheet) } else { $wb.Worksheets.Item(1) }
        $data = $ws.UsedRange.Value2
    }
    catch {
        throw "Could not read the report sheet from '$FullPath'. If it is open, make sure no cell is in edit mode. $($_.Exception.Message)"
    }
    finally {
        if ($ws) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ws) }
        if ($own) {
            if ($wb)    { try { $wb.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wb) }
            if ($excel) { try { $excel.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
        }
    }
    if ($data -isnot [object[,]]) { return $null }
    return ,$data
}

# ---------- Read the report ----------
Write-Log "Starting validation of the generated report against LIVE Azure tags (read-only, no files written)"
Write-Log "Report     : $fullReport"
Write-Log "Azure acct : $((Get-AzContext).Account.Id)"

$data = Get-SheetData -FullPath $fullReport -Sheet $SheetName
if (-not $data) { throw "The report sheet is empty." }

$col = @{}
for ($c = 1; $c -le $data.GetLength(1); $c++) {
    $h = (ConvertTo-CellString $data[1, $c]).Trim()
    if ($h -and -not $col.ContainsKey($h)) { $col[$h] = $c }
}
foreach ($h in (@("SubscriptionName", "SubscriptionId") + $TagNames)) {
    if (-not $col.ContainsKey($h)) { throw "Header '$h' not found in row 1 of the report. Headers found: $($col.Keys -join ', ')" }
}

$rows = for ($r = 2; $r -le $data.GetLength(0); $r++) {
    $name = (ConvertTo-CellString $data[$r, $col["SubscriptionName"]]).Trim()
    $id   = (ConvertTo-CellString $data[$r, $col["SubscriptionId"]]).Trim()
    if (-not $name -and -not $id) { continue }
    $vals = @{}
    foreach ($t in $TagNames) { $vals[$t] = ConvertTo-CellString $data[$r, $col[$t]] }   # exact, not trimmed
    [pscustomobject]@{ Row = $r; Name = $name; Id = $id; Values = $vals }
}
$rows = @($rows)
if ($rows.Count -eq 0) { throw "No subscription rows found in the report." }
Write-Log "Subscriptions in report: $($rows.Count)"

$dupes = $rows | Where-Object { $_.Id } | Group-Object { $_.Id.ToLower() } | Where-Object Count -gt 1
foreach ($d in $dupes) { Write-Log "WARNING: duplicate subscription ID in report: $($d.Name) ($($d.Count) rows)" "Yellow" }

# ---------- Live subscription names (read-only) ----------
$subById = @{}
try {
    foreach ($s in @(Get-AzSubscription -ErrorAction Stop)) { $subById[$s.Id.ToLower()] = $s.Name }
    Write-Log "Accessible subscriptions found in Azure: $($subById.Count)"
}
catch {
    Write-Log "WARNING: could not list subscriptions ($(($_.Exception.Message -split "`n")[0])). Subscription NAMES will not be verified; tags still are." "Yellow"
}

# ---------- Helpers ----------
function Get-SubscriptionTags {
    param([string]$SubId)
    for ($attempt = 1; $attempt -le $MaxRetries; $attempt++) {
        try {
            $t = Get-AzTag -ResourceId "/subscriptions/$SubId" -ErrorAction Stop
            return $t.Properties.TagsProperty
        }
        catch {
            $msg = $_.Exception.Message
            if ($msg -match "AuthorizationFailed|SubscriptionNotFound|InvalidAuthenticationToken|not found|Forbidden|does not have authorization") { throw }
            if ($attempt -eq $MaxRetries) { throw }
            Start-Sleep -Seconds (2 * $attempt)
        }
    }
}

function Get-TagValue {
    param($Tags, [string]$Name)
    if (-not $Tags) { return @{ Exists = $false; Value = "" } }
    foreach ($k in $Tags.Keys) {
        if ($k -ieq $Name) { return @{ Exists = $true; Value = "$($Tags[$k])" } }
    }
    return @{ Exists = $false; Value = "" }
}

function Compare-TagValue {
    param([string]$Report, [bool]$AzureExists, [string]$Azure)

    if ($Report -ceq $Azure) { return "MATCH" }          # includes: report blank + tag not set/empty in Azure
    if (-not $AzureExists)   { return "TAG_MISSING_IN_AZURE" }      # report has a value, Azure has no such tag
    if ([string]::IsNullOrEmpty($Report))  { return "BLANK_IN_REPORT" }   # Azure has a value, report cell blank
    if ([string]::IsNullOrEmpty($Azure))   { return "TAG_EMPTY_IN_AZURE" }
    if ($Report.Trim() -ceq $Azure.Trim()) { return "WHITESPACE_DIFFERENCE" }
    if ($Report.Trim() -ieq $Azure.Trim()) { return "CASE_DIFFERENCE" }
    return "VALUE_MISMATCH"
}

# ---------- Main loop ----------
$results = New-Object System.Collections.Generic.List[object]
$i = 0

foreach ($row in $rows) {
    $i++
    Write-Progress -Activity "Validating report against live Azure tags" -Status "$i of $($rows.Count): $($row.Name)" -PercentComplete (($i / $rows.Count) * 100)

    $issues = @()
    $err = $null
    $tags = $null

    $g = [guid]::Empty
    if (-not $row.Id)                                  { $err = "No subscription ID in the report row." }
    elseif (-not [guid]::TryParse($row.Id, [ref]$g))   { $err = "'$($row.Id)' is not a valid subscription ID." }
    else {
        try { $tags = Get-SubscriptionTags -SubId $row.Id }
        catch { $err = ($_.Exception.Message -split "`n")[0].Trim() }
    }

    if (-not $err) {
        # Subscription name check (only if the live name is known)
        $key = $row.Id.ToLower()
        if ($subById.ContainsKey($key) -and ($row.Name -cne $subById[$key])) {
            $issues += [pscustomobject]@{ Tag = "SubscriptionName"; Issue = "NAME_MISMATCH"; ReportValue = $row.Name; AzureValue = $subById[$key] }
        }
        foreach ($t in $TagNames) {
            $az     = Get-TagValue -Tags $tags -Name $t
            $status = Compare-TagValue -Report $row.Values[$t] -AzureExists $az.Exists -Azure $az.Value
            if ($status -ne "MATCH") {
                $issues += [pscustomobject]@{
                    Tag = $t; Issue = $status; ReportValue = $row.Values[$t]
                    AzureValue = if ($az.Exists) { $az.Value } else { "<tag not set>" }
                }
            }
        }
    }

    $overall = if ($err) { "ERROR" } elseif ($issues.Count -eq 0) { "MATCH" } else { "MISMATCH" }
    $detail  = if ($err) { $err } else { ($issues | ForEach-Object { "$($_.Tag): $($_.Issue)" }) -join "; " }

    $results.Add([pscustomobject]@{
        Row = $row.Row; Name = $row.Name; Id = $row.Id; Status = $overall; Detail = $detail; Issues = $issues; Error = $err
    })

    switch ($overall) {
        "MATCH"    { Write-Log "OK        $($row.Name) ($($row.Id))" }
        "ERROR"    { Write-Log "ERROR     $($row.Name) ($($row.Id)) - $err" "Red" }
        "MISMATCH" { Write-Log "MISMATCH  $($row.Name) ($($row.Id)) - $detail" "Yellow" }
    }
}
Write-Progress -Activity "Validating report against live Azure tags" -Completed

# ---------- Final summary (everything together) ----------
$total    = $results.Count
$matched  = @($results | Where-Object Status -eq "MATCH").Count
$mismatch = @($results | Where-Object Status -eq "MISMATCH")
$errored  = @($results | Where-Object Status -eq "ERROR")
$pct      = if ($total) { [math]::Round(($matched / $total) * 100, 2) } else { 0 }

Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  REPORT vs LIVE AZURE VALIDATION SUMMARY   (tags: $($TagNames -join ', '))"
Write-Plain ("=" * 78)
Write-Plain "  Run time              : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Plain "  Report file           : $fullReport"
Write-Plain "  Subscriptions checked : $total"
Write-Plain "  Matched live Azure    : $matched" "Green"
Write-Plain "  Mismatched            : $($mismatch.Count)" $(if ($mismatch.Count) { "Yellow" } else { "Green" })
Write-Plain "  Errors (not verified) : $($errored.Count)"  $(if ($errored.Count)  { "Red" }    else { "Green" })
Write-Plain "  Match rate            : $pct%"
Write-Plain ("-" * 78)

if ($total -gt 0 -and $matched -eq $total) {
    Write-Plain "  RESULT: 100% MATCH - every name and tag value in the report equals the live Azure subscription." "Green"
}
else {
    Write-Plain "  RESULT: NOT A 100% MATCH - $($mismatch.Count + $errored.Count) subscription(s) need attention." "Red"

    if ($mismatch.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  Issue breakdown:"
        $mismatch | ForEach-Object { $_.Issues } | Group-Object Tag, Issue | Sort-Object Name |
            ForEach-Object { Write-Plain ("    {0,-40} {1}" -f $_.Name, $_.Count) }

        Write-Plain ""
        Write-Plain "  MISMATCHED SUBSCRIPTIONS ($($mismatch.Count)):" "Yellow"
        foreach ($m in $mismatch) {
            Write-Plain ""
            Write-Plain "  * $($m.Name)   (report row $($m.Row))"
            Write-Plain "    ID: $($m.Id)"
            foreach ($iss in $m.Issues) {
                Write-Plain ("    - {0,-16} {1,-22} Report='{2}'  Azure='{3}'" -f $iss.Tag, $iss.Issue, $iss.ReportValue, $iss.AzureValue)
            }
        }
    }

    if ($errored.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  COULD NOT BE VERIFIED ($($errored.Count)) - check access / tenant / sign-in:" "Red"
        foreach ($e in $errored) {
            Write-Plain "  * $($e.Name)   (report row $($e.Row))  [$($e.Id)]"
            Write-Plain "    $($e.Error)"
        }
    }
}
Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  No files were created or modified. Report workbook left untouched."
Write-Plain "  Azure: read-only."
Write-Plain ("=" * 78)
