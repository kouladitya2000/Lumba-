$ErrorActionPreference = "Stop"

$srcName = "YourFile.xlsx"                         # file name on your Desktop
$src     = Join-Path "$env:USERPROFILE\Desktop" $srcName
$outDir  = Join-Path "$env:USERPROFILE\Desktop" "ITSOwiseExcel"

if (-not (Test-Path $src)) { throw "Source file not found: $src" }
if (-not (Test-Path $outDir)) { [void](New-Item -ItemType Directory -Path $outDir) }
Write-Host "Source: $src"
Write-Host "Output: $outDir"

$excel = New-Object -ComObject Excel.Application
$excel.Visible = $false
$excel.DisplayAlerts = $false
$ok = 0; $fail = 0

try {
    $wb   = $excel.Workbooks.Open($src)
    $ws   = $wb.Worksheets.Item("Sheet1")
    $last = $ws.Cells.Item($ws.Rows.Count, 1).End(-4162).Row
    Write-Host "Last row: $last"

    $data   = $ws.Range("A2:G$last").Value2
    $groups = @{}
    $order  = New-Object System.Collections.ArrayList

    for ($i = 1; $i -le ($last - 1); $i++) {
        $itso = ("" + $data[$i,2]).Trim()
        if ($itso -eq "") { continue }
        $key = $itso.ToLower()
        if (-not $groups.ContainsKey($key)) {
            $groups[$key] = New-Object System.Collections.ArrayList
            [void]$order.Add($key)
        }
        [void]$groups[$key].Add($i)
    }
    Write-Host "ITSOs found: $($order.Count)"

    foreach ($key in $order) {
        $nb = $null
        try {
            $rows = $groups[$key]
            $n    = $rows.Count
            $arr  = New-Object 'object[,]' ($n + 1), 6
            $arr[0,0]="ITSO"; $arr[0,1]="SUBSCRIPTION NAME"; $arr[0,2]="ITSO DELEGATE"
            $arr[0,3]="EIM";  $arr[0,4]="EHC ID";            $arr[0,5]="BA ID"
            for ($r = 0; $r -lt $n; $r++) {
                $i = $rows[$r]
                $arr[($r+1),0] = $key
                $arr[($r+1),1] = $data[$i,1]
                $arr[($r+1),2] = $data[$i,3]
                $arr[($r+1),3] = $data[$i,5]
                $arr[($r+1),4] = $data[$i,6]
                $arr[($r+1),5] = $data[$i,7]
            }

            $nb = $excel.Workbooks.Add()
            $s  = $nb.Worksheets.Item(1)
            $s.Range($s.Cells.Item(1,1), $s.Cells.Item($n+1,6)).Value2 = $arr
            $s.Rows.Item(1).Font.Bold = $true
            [void]$s.Columns.AutoFit()

            $safe = ($key -replace '[\\/:*?"<>|]', '_')
            $file = Join-Path $outDir ($safe + ".xlsx")
            if (Test-Path $file) { Remove-Item $file -Force }
            $nb.SaveAs($file, 51)
            $nb.Close($false)
            $nb = $null

            if (Test-Path $file) { $ok++; Write-Host "Saved: $safe.xlsx" }
            else { $fail++; Write-Host "NOT FOUND after save: $file" -ForegroundColor Red }
        }
        catch {
            $fail++
            Write-Host "FAILED for $key : $($_.Exception.Message)" -ForegroundColor Red
            if ($nb -ne $null) { try { $nb.Close($false) } catch {} }
        }
    }
    $wb.Close($false)
}
catch {
    Write-Host "ERROR: $($_.Exception.Message)" -ForegroundColor Red
}
finally {
    $excel.Quit()
    [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($excel)
}

$onDisk = @(Get-ChildItem $outDir -Filter *.xlsx -ErrorAction SilentlyContinue).Count
Write-Host "Saved OK: $ok | Failed: $fail | Files actually on disk: $onDisk" -ForegroundColor Cyan
Write-Host "Open this folder to check: $outDir"
