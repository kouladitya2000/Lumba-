# ---- Settings (edit these) ----
$inputCsv  = "C:\Path\report.csv"                       # the validated report saved as CSV
$outputCsv = "C:\Path\EHC_Output_" + (Get-Date -Format "yyyyMMdd_HHmmss") + ".csv"
$baseUrl   = "https://cstmg-fa-euw-01.cstmg-ase-euw-01.appserviceenvironment.net"
$faCode    = "PASTE_YOUR_EXISTING_FACODE_VALUE_HERE"    # same value you already use in your API script
$processStatuses = @('BAD','MULTI','SKIPPED')           # only rows with these Status values are processed

# ---- Load the report ----
$rows = @(Import-Csv $inputCsv -Encoding UTF8 -WarningAction SilentlyContinue)
if ($rows.Count -eq 0) { throw "Input CSV loaded 0 rows - check the path" }
$cols = $rows[0].PSObject.Properties.Name
foreach ($need in 'SubscriptionName','SubscriptionId','BA ID','Status') {
    if ($cols -notcontains $need) { throw "Column '$need' not found. Headers seen: $($cols -join ', ')" }
}

# Wraps a value in quotes for CSV output
function Q($v) { '"' + ([string]$v -replace '"', '""') + '"' }

# ---- Finds the ehcInfo list wherever it sits in the JSON ----
function Find-Ehc($o) {
    if ($null -eq $o) { return $null }
    if ($o -is [array]) {
        foreach ($i in $o) { $r = Find-Ehc $i; if ($null -ne $r) { return $r } }
        return $null
    }
    if ($o -is [pscustomobject]) {
        foreach ($p in $o.PSObject.Properties) {
            if ($p.Name -eq 'ehcInfo') { return @{ Items = $p.Value } }
        }
        foreach ($p in $o.PSObject.Properties) {
            $r = Find-Ehc $p.Value
            if ($null -ne $r) { return $r }
        }
    }
    return $null
}

# ---- Calls the API for one BA ID and builds the EHC text for column D ----
function Get-EhcText($ba) {
    $url = $baseUrl + "/api/GetBAInfo?baid=" + $ba + "&" + $faCode
    $resp = $null
    $err = ""
    for ($try = 1; $try -le 2; $try++) {
        try {
            $resp = Invoke-WebRequest -UseBasicParsing -Uri $url -Method GET -ContentType "application/json"
            break
        }
        catch {
            $err = $_.Exception.Message
            if ($err -like '*404*') { return "BA NOT FOUND (HTTP 404)" }
            if ($err -like '*401*' -or $err -like '*403*') { throw "Access denied ($err) - check your faCode" }
            Start-Sleep -Seconds 2
        }
    }
    if ($null -eq $resp) { return "API ERROR: $err" }

    try { $j = $resp.Content | ConvertFrom-Json }
    catch { return "API ERROR: response was not valid JSON" }

    $ehc = $j.busappInfo.ehcInfo
    if (-not $ehc) { $ehc = $j.ehcInfo }
    if (-not $ehc) {
        $found = Find-Ehc $j
        if ($found) { $ehc = $found.Items }
    }
    if (-not $ehc) { return "NO EHC FOUND" }

    $mpc = @(); $wd = @(); $other = @()
    foreach ($e in @($ehc)) {
        $id = ([string]$e.ehcId).Trim()
        $st = ([string]$e.ehcStatus).Trim()
        if     ($st -eq 'Move to Project Control') { $mpc   += $id }
        elseif ($st -eq 'Withdrawn')               { $wd    += "$id(Withdrawn)" }
        else                                       { $other += "$id($st)" }
    }

    if     ($mpc.Count   -gt 0) { return ($mpc -join ", ") }
    elseif ($wd.Count    -gt 0) { return ($wd -join ", ") }
    elseif ($other.Count -gt 0) { return ($other -join ", ") }
    else                        { return "NO EHC FOUND" }
}

# ---- Process rows (API is called once per unique BA ID) ----
$cache   = @{}
$lines   = @()
$ehcVals = @()
$skipped = @{}
$n = 0

$lines += (Q 'SubscriptionName') + "," + (Q 'SubscriptionId') + "," + (Q 'BA ID') + "," + (Q 'EHC ID')

foreach ($r in $rows) {
    $n++
    $sub    = ([string]$r.'SubscriptionName').Trim()
    $subId  = ([string]$r.'SubscriptionId').Trim()
    $baRaw  = ([string]$r.'BA ID').Trim()
    $ba     = $baRaw -replace '\s', ''
    $status = ([string]$r.'Status').Trim().ToUpper()

    if (-not $sub -and -not $subId -and -not $ba) { continue }   # skip blank rows

    # Only BAD / MULTI / SKIPPED rows are processed
    if ($processStatuses -notcontains $status) {
        $key = "(blank)"
        if ($status) { $key = $status }
        if ($skipped.ContainsKey($key)) { $skipped[$key] = $skipped[$key] + 1 } else { $skipped[$key] = 1 }
        continue
    }

    if (-not $cache.ContainsKey($ba)) {
        $cache[$ba] = Get-EhcText $ba
        Write-Host "[$n/$($rows.Count)] $ba ($status) -> $($cache[$ba])"
    }

    $ehcVals += $cache[$ba]
    $lines   += (Q $sub) + "," + (Q $subId) + "," + (Q $baRaw) + "," + (Q $cache[$ba])
}

if ($ehcVals.Count -eq 0) { throw "No rows with Status in ($($processStatuses -join ', ')) were found - nothing to write" }
$lines | Out-File -FilePath $outputCsv -Encoding UTF8

# ---- Summary ----
$apiErr = 0; $nf404 = 0; $none = 0; $wdRows = 0
foreach ($v in $ehcVals) {
    if     ($v -like 'API ERROR*')       { $apiErr++ }
    elseif ($v -like 'BA NOT FOUND*')    { $nf404++ }
    elseif ($v -eq 'NO EHC FOUND')       { $none++ }
    elseif ($v -like '*(Withdrawn)*')    { $wdRows++ }
}

Write-Host "`n===== SUMMARY =====" -ForegroundColor Cyan
Write-Host "Rows in input        : $($rows.Count)"
Write-Host "Rows processed       : $($ehcVals.Count)"
foreach ($k in $skipped.Keys) { Write-Host ("Skipped (" + $k + ")        : " + $skipped[$k]) }
Write-Host "Unique BA IDs hit    : $($cache.Count)"
Write-Host "Withdrawn fallback   : $wdRows rows"
Write-Host "BA not found (404)   : $nf404 rows"
Write-Host "No EHC found         : $none rows"
Write-Host "API errors           : $apiErr rows"
Write-Host "Output file          : $outputCsv"
if ($apiErr -gt 0) { Write-Host "Some rows hit real API errors - search the output for 'API ERROR' and re-run." -ForegroundColor Red }
