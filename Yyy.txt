<#
.SYNOPSIS
  Compares BA ID and EIM between two Excel files. Log check only, output on screen.

  FILE 1 (-ReportFile)  : the report generated by the tag script
                          (SubscriptionName | SubscriptionId | BA ID | EIM | EHC ID)
  FILE 2 (-MappingFile) : the other file
                          (SubscriptionName | SubscriptionId | BA ID | EIM)

  Rows are matched on SubscriptionId (primary key). If a row has no ID, it is matched
  on SubscriptionName instead. EHC ID is ignored (it does not exist in file 2).

.READ-ONLY GUARANTEE
  - Both workbooks are opened read-only and closed without saving.
  - Nothing is written anywhere (no report, no log file). Everything prints on screen.
  - No Azure calls at all.

.USAGE
  1. Set -ReportFile and -MappingFile below (or pass them as parameters).
  2. Run. Add -MismatchesOnly to hide the per-row OK lines.

  Values are compared after trimming spaces, ignoring upper/lower case.
  A value such as "NF" in file 2 is still reported as a mismatch, but is labelled as a
  placeholder so you can tell it apart from a genuinely different value.
#>

[CmdletBinding()]
param(
    # >>> SET THESE TWO PATHS <<<
    [string]$ReportFile  = "C:\Users\45270978\Desktop\TagsValidator\<generated-report-file>.xlsx",
    [string]$MappingFile = "C:\Users\45270978\Desktop\TagsValidator\<mapping-file>.xlsx",

    # Leave blank to use the first sheet of each workbook
    [string]$ReportSheet  = "",
    [string]$MappingSheet = "",

    [string]$NameHeader = "SubscriptionName",
    [string]$IdHeader   = "SubscriptionId",
    [string[]]$CompareHeaders = @("BA ID", "EIM"),

    # Values in file 2 that mean "not found" rather than a real value
    [string[]]$PlaceholderValues = @("NF"),

    [switch]$MismatchesOnly
)

$ErrorActionPreference = "Stop"

function Write-Log {
    param([string]$Message, [string]$Color = "")
    $line = "[{0}] {1}" -f (Get-Date -Format "HH:mm:ss"), $Message
    if ($Color) { Write-Host $line -ForegroundColor $Color } else { Write-Host $line }
}
function Write-Plain {
    param([string]$Message = "", [string]$Color = "")
    if ($Color) { Write-Host $Message -ForegroundColor $Color } else { Write-Host $Message }
}

# ---------- Pre-flight ----------
foreach ($f in @(@("ReportFile", $ReportFile), @("MappingFile", $MappingFile))) {
    if (-not (Test-Path -LiteralPath $f[1])) { throw "-$($f[0]) not found: $($f[1])  (set the correct path at the top of the script)" }
}
$fullReport  = (Resolve-Path -LiteralPath $ReportFile).Path
$fullMapping = (Resolve-Path -LiteralPath $MappingFile).Path

# ---------- Excel helpers (read-only) ----------
function ConvertTo-CellString($v) {
    if ($null -eq $v) { return "" }
    if ($v -is [double]) { return $v.ToString("0.##########", [Globalization.CultureInfo]::InvariantCulture) }
    return [string]$v
}

function Read-SheetData {
    param([string]$FullPath, [string]$Sheet)

    $excel = $null; $wb = $null; $ws = $null; $ur = $null
    $own = $false; $running = $null; $data = $null

    try { $running = [Runtime.InteropServices.Marshal]::GetActiveObject("Excel.Application") } catch { $running = $null }
    if ($running) {
        foreach ($w in $running.Workbooks) {
            if ($w.FullName -ieq $FullPath) { $wb = $w; $excel = $running; break }
        }
    }

    try {
        if (-not $wb) {
            $excel = New-Object -ComObject Excel.Application
            $excel.Visible = $false
            $excel.DisplayAlerts = $false
            $own = $true
            $wb = $excel.Workbooks.Open($FullPath, 0, $true)      # UpdateLinks=0, ReadOnly=true
        }
        $ws = if ($Sheet) { $wb.Worksheets.Item($Sheet) } else { $wb.Worksheets.Item(1) }
        $ur = $ws.UsedRange
        $data = $ur.Value2
    }
    catch {
        throw "Could not read '$FullPath' (sheet '$Sheet'). $($_.Exception.Message)"
    }
    finally {
        if ($ur) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ur) }
        if ($ws) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ws) }
        if ($own) {
            if ($wb)    { try { $wb.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wb) }
            if ($excel) { try { $excel.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
        }
    }
    return ,$data
}

function Import-Rows {
    param([string]$Path, [string]$Sheet, [string]$Label)

    $data = Read-SheetData -FullPath $Path -Sheet $Sheet
    if ($data -isnot [object[,]]) { throw "$Label : the sheet is empty." }

    $col = @{}
    for ($c = 1; $c -le $data.GetLength(1); $c++) {
        $h = (ConvertTo-CellString $data[1, $c]).Trim()
        if ($h -and -not $col.ContainsKey($h)) { $col[$h] = $c }
    }
    foreach ($h in (@($NameHeader, $IdHeader) + $CompareHeaders)) {
        if (-not $col.ContainsKey($h)) {
            throw "$Label : header '$h' not found in row 1. Headers found: $($col.Keys -join ', ')"
        }
    }

    $rows = New-Object System.Collections.Generic.List[object]
    for ($r = 2; $r -le $data.GetLength(0); $r++) {
        $name = (ConvertTo-CellString $data[$r, $col[$NameHeader]]).Trim()
        $id   = (ConvertTo-CellString $data[$r, $col[$IdHeader]]).Trim()
        if (-not $name -and -not $id) { continue }
        $vals = @{}
        foreach ($h in $CompareHeaders) { $vals[$h] = (ConvertTo-CellString $data[$r, $col[$h]]).Trim() }
        $rows.Add([pscustomobject]@{ Row = $r; Name = $name; Id = $id; Values = $vals })
    }
    return ,$rows
}

# ---------- Read both files ----------
Write-Log "Starting comparison (read-only, nothing is written)"
Write-Log "File 1 (report)  : $fullReport"
Write-Log "File 2 (mapping) : $fullMapping"
Write-Log "Comparing        : $($CompareHeaders -join ', ')   (key: $IdHeader, fallback: $NameHeader)"

$rowsA = Import-Rows -Path $fullReport  -Sheet $ReportSheet  -Label "File 1 (report)"
$rowsB = Import-Rows -Path $fullMapping -Sheet $MappingSheet -Label "File 2 (mapping)"
Write-Log "Rows read: file 1 = $($rowsA.Count), file 2 = $($rowsB.Count)"

# ---------- Index file 2 ----------
$bById   = @{}
$bByName = @{}
foreach ($b in $rowsB) {
    if ($b.Id) {
        $k = $b.Id.ToLower()
        if ($bById.ContainsKey($k)) { Write-Log "WARNING: duplicate subscription ID in file 2: $($b.Id) (row $($b.Row)); first occurrence is used." "Yellow" }
        else { $bById[$k] = $b }
    }
    if ($b.Name) {
        $nk = $b.Name.ToLower()
        if (-not $bByName.ContainsKey($nk)) { $bByName[$nk] = New-Object System.Collections.Generic.List[object] }
        $bByName[$nk].Add($b)
    }
}
$seenA = @{}
foreach ($a in $rowsA) {
    if ($a.Id) {
        $k = $a.Id.ToLower()
        if ($seenA.ContainsKey($k)) { Write-Log "WARNING: duplicate subscription ID in file 1: $($a.Id) (row $($a.Row))" "Yellow" }
        $seenA[$k] = $true
    }
}

# ---------- Compare ----------
$usedB        = @{}      # file 2 rows that were matched
$matchedAll   = New-Object System.Collections.Generic.List[object]
$mismatches   = New-Object System.Collections.Generic.List[object]
$notInB       = New-Object System.Collections.Generic.List[object]
$perHeader    = @{}
foreach ($h in $CompareHeaders) { $perHeader[$h] = @{ Same = 0; Diff = 0; Placeholder = 0 } }
$byNameCount  = 0
$i = 0

foreach ($a in $rowsA) {
    $i++
    Write-Progress -Activity "Comparing" -Status "$i of $($rowsA.Count): $($a.Name)" -PercentComplete (($i / $rowsA.Count) * 100)

    $b = $null; $how = ""; $problem = $null
    if ($a.Id -and $bById.ContainsKey($a.Id.ToLower())) { $b = $bById[$a.Id.ToLower()]; $how = "id" }
    elseif (-not $a.Id -and $a.Name -and $bByName.ContainsKey($a.Name.ToLower())) {
        $cand = $bByName[$a.Name.ToLower()]
        if ($cand.Count -eq 1) { $b = $cand[0]; $how = "name"; $byNameCount++ }
        else { $problem = "name matches $($cand.Count) rows in file 2; add the subscription ID to file 1" }
    }

    $label = "$($a.Name) ($($a.Id))"

    if (-not $b) {
        $reason = if ($problem) { $problem } else { "not found in file 2" }
        $notInB.Add([pscustomobject]@{ Row = $a.Row; Name = $a.Name; Id = $a.Id; Reason = $reason })
        Write-Log "NOT FOUND $label - $reason" "Yellow"
        continue
    }

    $usedB[$b.Row] = $true
    $diffs = @()
    foreach ($h in $CompareHeaders) {
        $va = $a.Values[$h]; $vb = $b.Values[$h]
        if ($va -ieq $vb) { $perHeader[$h].Same++ }
        else {
            $isPh = $false
            foreach ($p in $PlaceholderValues) { if ($vb -ieq $p) { $isPh = $true } }
            $perHeader[$h].Diff++
            if ($isPh) { $perHeader[$h].Placeholder++ }
            $diffs += [pscustomobject]@{ Header = $h; Report = $va; Mapping = $vb; Placeholder = $isPh }
        }
    }

    if ($diffs.Count -eq 0) {
        $matchedAll.Add($a)
        if (-not $MismatchesOnly) { Write-Log "OK        $label" }
    }
    else {
        $mismatches.Add([pscustomobject]@{ RowA = $a.Row; RowB = $b.Row; Name = $a.Name; Id = $a.Id; Diffs = $diffs })
        Write-Log "MISMATCH  $label" "Red"
        foreach ($d in $diffs) {
            $note = if ($d.Placeholder) { "   <- file 2 has placeholder '$($d.Mapping)'" } else { "" }
            Write-Log ("            {0,-6}: file1 = '{1}'   file2 = '{2}'{3}" -f $d.Header, $d.Report, $d.Mapping, $note) "Red"
        }
    }
}
Write-Progress -Activity "Comparing" -Completed

$onlyInB = @($rowsB | Where-Object { -not $usedB.ContainsKey($_.Row) })

# ---------- Summary ----------
$matchedKey = $rowsA.Count - $notInB.Count
Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  COMPARISON SUMMARY   (compared: $($CompareHeaders -join ', '))"
Write-Plain ("=" * 78)
Write-Plain "  Run time                         : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Plain "  File 1 (report)                  : $fullReport"
Write-Plain "  File 2 (mapping)                 : $fullMapping"
Write-Plain "  Rows in file 1 / file 2          : $($rowsA.Count) / $($rowsB.Count)"
Write-Plain "  Matched by key                   : $matchedKey$(if ($byNameCount) { "  ($byNameCount matched by name)" })"
Write-Plain "  Identical in all compared columns: $($matchedAll.Count)" "Green"
Write-Plain "  With one or more differences     : $($mismatches.Count)" $(if ($mismatches.Count) { "Red" } else { "Green" })
Write-Plain "  In file 1 but not in file 2      : $($notInB.Count)"   $(if ($notInB.Count) { "Yellow" } else { "Green" })
Write-Plain "  In file 2 but not in file 1      : $($onlyInB.Count)"  $(if ($onlyInB.Count) { "Yellow" } else { "Green" })
Write-Plain ("-" * 78)
Write-Plain "  Per column (of $matchedKey matched rows):"
foreach ($h in $CompareHeaders) {
    $ph = $perHeader[$h]
    Write-Plain ("    {0,-8} same: {1,-5} different: {2,-5} (of which file 2 placeholder: {3})" -f $h, $ph.Same, $ph.Diff, $ph.Placeholder)
}
Write-Plain ("-" * 78)

if ($mismatches.Count -eq 0 -and $notInB.Count -eq 0 -and $onlyInB.Count -eq 0) {
    Write-Plain "  RESULT: files are fully consistent for $($CompareHeaders -join ' and ')." "Green"
}
else {
    Write-Plain "  RESULT: DIFFERENCES FOUND - details below." "Red"

    if ($mismatches.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  MISMATCHES ($($mismatches.Count)):" "Red"
        foreach ($m in $mismatches) {
            Write-Plain ""
            Write-Plain "  * $($m.Name)   (file 1 row $($m.RowA), file 2 row $($m.RowB))"
            Write-Plain "    ID : $($m.Id)"
            foreach ($d in $m.Diffs) {
                $note = if ($d.Placeholder) { "   [file 2 placeholder]" } else { "" }
                Write-Plain ("    {0,-6}: file1 = '{1}'   file2 = '{2}'{3}" -f $d.Header, $d.Report, $d.Mapping, $note)
            }
        }
    }

    if ($notInB.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  IN FILE 1 BUT NOT FOUND IN FILE 2 ($($notInB.Count)):" "Yellow"
        foreach ($n in $notInB) {
            Write-Plain "  * $($n.Name)   (file 1 row $($n.Row))  [$($n.Id)]"
            Write-Plain "    $($n.Reason)"
        }
    }

    if ($onlyInB.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  IN FILE 2 BUT NOT IN FILE 1 ($($onlyInB.Count)):" "Yellow"
        foreach ($o in $onlyInB) {
            $vals = @(); foreach ($h in $CompareHeaders) { $vals += "$h=$($o.Values[$h])" }
            Write-Plain "  * $($o.Name)   (file 2 row $($o.Row))  [$($o.Id)]"
            Write-Plain "    $($vals -join ',  ')"
        }
    }
}

Write-Plain ""
Write-Plain ("-" * 78)
Write-Plain "  Read-only: no files written or changed, no Azure calls."
Write-Plain ("=" * 78)
