
$StartDate = Get-Date "2026-03-01"
$OutputFile = "$env:USERPROFILE\Desktop\File_Inventory.csv"

$Extensions = @("*.xlsx", "*.xls", "*.xlsm", "*.xlsb", "*.cab")

$Results = foreach ($Drive in (Get-PSDrive -PSProvider FileSystem)) {

    Get-ChildItem -Path $Drive.Root `
        -Include $Extensions `
        -File `
        -Recurse `
        -ErrorAction SilentlyContinue |
    Where-Object {
        $_.LastWriteTime -ge $StartDate
    } |
    Select-Object `
        @{Name="File name"; Expression={$_.Name}},
        @{Name="Date"; Expression={$_.LastWriteTime.ToString("dd-MMM-yyyy HH:mm")}},
        @{Name="Path"; Expression={$_.DirectoryName}}
}

$Results |
    Sort-Object Date -Descending |
    Export-Csv -Path $OutputFile -NoTypeInformation -Encoding UTF8

Write-Host "Completed! Report saved to $OutputFile"
