# ============ SETTINGS ============
$baseUrl = "https://cstmg-fa-euw-01.cstmg-ase-euw-01.appserviceenvironment.net"

#>>> PASTE 1: your function key here (keep the "code=" part)
$faCode  = "code=<YOUR_FUNCTION_KEY>"

$maxAttempts  = 4     # tries per BA+RITM pair
$pauseBetween = 500   # ms pause between calls (raise to 1000-2000 if still blocked)

# ============ INPUT LIST ============
#>>> PASTE 2: paste your whole BA + RITM list between the two markers below,
#>>> exactly as it appears in Notepad++ (BA line, then its RITMs, NONE lines are fine).
#>>> The closing "@ MUST stay alone at the very start of its line, no spaces before it.
$raw = @"
BA0017001
NONE
BA0013517
RITM32950187,RITM19196554,RITM19859855
BA0011049
RITM32945073,RITM27789439
"@

# ============ BUILD BA -> RITM PAIRS (do not edit) ============
$entries = @()
foreach ($line in ($raw -split "\r?\n")) {
    if ($line -match 'BA\d+') { $entries += @{ BA = $matches[0]; RITMs = @() } }
    if ($entries.Count -gt 0) {
        $found = Select-String -InputObject $line -Pattern 'RITM\d+' -AllMatches
        if ($found) { $entries[-1].RITMs += $found.Matches.Value }
    }
}

# ============ API TEST WITH RETRIES (do not edit) ============
function Test-Pair($ba, $ritm) {
    $url = "$baseUrl/api/GetBAData?baid=$ba&ehcId=$ritm&$faCode"
    for ($i = 1; $i -le $maxAttempts; $i++) {
        try {
            $resp = Invoke-WebRequest -UseBasicParsing -Uri $url -Method GET -ContentType "application/json"
            if ($resp.StatusCode -eq 200) { return $true }
        }
        catch { }
        if ($i -lt $maxAttempts) { Start-Sleep -Seconds (3 * $i) }   # waits 3s, 6s, 9s
    }
    return $false
}

# ============ RUN (do not edit) ============
$cache   = @{}
$results = foreach ($e in $entries) {
    $working = @()
    foreach ($r in $e.RITMs) {
        $key = "$($e.BA)|$r"
        if (-not $cache.ContainsKey($key)) {
            $cache[$key] = Test-Pair $e.BA $r
            Start-Sleep -Milliseconds $pauseBetween
            Write-Host "$($e.BA) $r -> $($cache[$key])"
        }
        if ($cache[$key]) { $working += $r }
    }
    $txt = if ($working.Count -gt 0) { $working -join "," } else { "NONE WORKED" }
    New-Object PSObject -Property @{ BAID = $e.BA; WorkingEHCs = $txt }
}

# ============ OUTPUT ============
$results | Select-Object BAID, WorkingEHCs | Format-Table -AutoSize
$results | Select-Object BAID, WorkingEHCs | Export-Csv "$env:USERPROFILE\Desktop\BA_WorkingEHC.csv" -NoTypeInformation
