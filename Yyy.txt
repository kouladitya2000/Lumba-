<#
.SYNOPSIS
  Calls the BA / EHC API (GET) for every report row and writes a NEW Excel file.

  INPUT : the generated report .xlsx
          (SubscriptionName | SubscriptionId | BA ID | EIM | EHC ID)
  OUTPUT: a new .xlsx with those five columns + a 6th column "Notes".
          Row colours:
            GREEN  = API returned data for this BA ID + EHC ID               (good)
            RED    = API returned no data / an error                         (bad)
            YELLOW = not checked: no EHC ID in the report                    (raise with SRE)
            BLUE   = this row came from a cell holding SEVERAL comma-separated
                     EHC IDs. The cell is split into one row per EHC ID and each
                     ID is checked separately.

  NOTES column
    - green row : "OK"
    - red row   : the error (e.g. "HTTP 404 - Not Found (no data)")
    - yellow row: "No EHC ID in report ... not checked"
    - blue rows : "N comma-separated EHC IDs found"
  For blue rows the result of each individual ID (and any mixed result, e.g. one ID
  returns data and the other does not) is shown in the PowerShell log ONLY.

.READ-ONLY GUARANTEE
  - API   : HTTP GET only (the same call you ran by hand). Nothing is created or changed.
  - Azure : no Azure / portal calls at all (no Az cmdlets, no sign-in needed).
  - Input : the report is opened read-only and closed without saving.
  - The ONLY file written is the new .xlsx (never overwrites an existing file).
  - The function key is never printed and never written to the output file.

.HOW A ROW IS JUDGED
  - HTTP 200 with a non-empty body  -> GOOD
  - HTTP 404 / any other error, or an empty body -> BAD
  - EHC ID cell is "Empty Tag" / "Tag does not exist" / blank -> not sent, YELLOW
  - BA ID blank -> not sent, BAD
  - 5xx / 429 / timeouts are retried (-MaxRetries). 404 and other 4xx are not retried.
  - If 3 calls in a row come back 401/403 the script STOPS (wrong key / no access).

.SENSITIVITY LABEL
  The output workbook is opened VISIBLE so the "Internal" label prompt can show.
  Click "Internal" when asked, or pass -LabelId and -TenantId to set it automatically.

.USAGE
  .\Validate-BaEhcApi.ps1
  You are prompted for the API base URL and the function key (typed hidden).
  Enter only the key value, i.e. the part AFTER "code=".
#>

[CmdletBinding()]
param(
    # >>> SET THE REPORT PATH HERE <<<
    [string]$ReportFile = "C:\Users\45270978\Desktop\TagsValidator\Stage2SubscriptionTagsReportEHCValidateHelper_20261009_160619.xlsx",
    [string]$SheetName  = "",                       # blank = first sheet

    # API (prompted if left blank). Do not hard-code the key in this file.
    [string]$BaseUrl = "",
    [string]$ApiPath = "/api/GetBAData",
    [securestring]$FunctionKey,

    # Leave blank to save next to the report as <reportname>_ApiValidated_<timestamp>.xlsx
    [string]$OutputPath = "",

    # EHC ID cell values that mean "no EHC ID" -> row is yellow and not sent to the API
    [string[]]$NoEhcValues = @("Empty Tag", "Tag does not exist"),

    [int]$DelayMs    = 200,     # pause between API calls
    [int]$TimeoutSec = 60,
    [int]$MaxRetries = 3,

    # Sensitivity label (optional - leave blank to click "Internal" in Excel's prompt)
    [string]$LabelId   = "",
    [string]$TenantId  = "",
    [string]$LabelName = "Internal"
)

$ErrorActionPreference = "Stop"
try { [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12 } catch {}

function Write-Log {
    param([string]$Message, [string]$Color = "")
    $line = "[{0}] {1}" -f (Get-Date -Format "HH:mm:ss"), $Message
    if ($Color) { Write-Host $line -ForegroundColor $Color } else { Write-Host $line }
}
function Write-Plain {
    param([string]$Message = "", [string]$Color = "")
    if ($Color) { Write-Host $Message -ForegroundColor $Color } else { Write-Host $Message }
}

# ---------- Pre-flight ----------
if (-not (Test-Path -LiteralPath $ReportFile)) {
    throw "Report file not found: $ReportFile  (set the correct path at the top of the script)"
}
$fullReport = (Resolve-Path -LiteralPath $ReportFile).Path

if (-not $BaseUrl) { $BaseUrl = Read-Host "API base URL (e.g. https://<app>.appserviceenvironment.net)" }
$BaseUrl = $BaseUrl.Trim().TrimEnd('/')
if ($BaseUrl -notmatch '^https?://') { throw "BaseUrl must start with http:// or https://" }

if (-not $FunctionKey) { $FunctionKey = Read-Host "Function key (the part after 'code=')" -AsSecureString }
$bstr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($FunctionKey)
try { $key = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($bstr) }
finally { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($bstr) }
$key = ($key.Trim() -replace '^code=', '')
if (-not $key) { throw "No function key supplied." }

$stamp = Get-Date -Format "yyyyMMdd_HHmmss"
if (-not $OutputPath) {
    $base = [IO.Path]::GetFileNameWithoutExtension($fullReport)
    $OutputPath = Join-Path (Split-Path -Parent $fullReport) "${base}_ApiValidated_$stamp.xlsx"
}
if ([IO.Path]::GetExtension($OutputPath).ToLower() -ne ".xlsx") { throw "-OutputPath must end with .xlsx" }
$outDir = Split-Path -Parent $OutputPath
if ($outDir -and -not (Test-Path -LiteralPath $outDir)) { throw "Output folder does not exist: $outDir" }
if (Test-Path -LiteralPath $OutputPath) { throw "Output file already exists (not overwriting): $OutputPath" }

# ---------- Excel read helper (read-only) ----------
function ConvertTo-CellString($v) {
    if ($null -eq $v) { return "" }
    if ($v -is [double]) { return $v.ToString("0.##########", [Globalization.CultureInfo]::InvariantCulture) }
    return [string]$v
}

function Read-SheetData {
    param([string]$FullPath, [string]$Sheet)
    $excel = $null; $wb = $null; $ws = $null; $ur = $null
    $own = $false; $running = $null; $data = $null

    try { $running = [Runtime.InteropServices.Marshal]::GetActiveObject("Excel.Application") } catch { $running = $null }
    if ($running) {
        foreach ($w in $running.Workbooks) {
            if ($w.FullName -ieq $FullPath) { $wb = $w; $excel = $running; break }
        }
    }
    try {
        if (-not $wb) {
            $excel = New-Object -ComObject Excel.Application
            $excel.Visible = $false
            $excel.DisplayAlerts = $false
            $own = $true
            $wb = $excel.Workbooks.Open($FullPath, 0, $true)      # UpdateLinks=0, ReadOnly=true
        }
        $ws = if ($Sheet) { $wb.Worksheets.Item($Sheet) } else { $wb.Worksheets.Item(1) }
        $ur = $ws.UsedRange
        $data = $ur.Value2
    }
    catch { throw "Could not read '$FullPath'. $($_.Exception.Message)" }
    finally {
        if ($ur) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ur) }
        if ($ws) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ws) }
        if ($own) {
            if ($wb)    { try { $wb.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wb) }
            if ($excel) { try { $excel.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
        }
    }
    return ,$data
}

# ---------- Read the report ----------
Write-Log "Starting BA/EHC API validation (GET only; no Azure/portal calls)"
Write-Log "Report file : $fullReport"
Write-Log "API         : $BaseUrl$ApiPath   (function key hidden)"

$data = Read-SheetData -FullPath $fullReport -Sheet $SheetName
if ($data -isnot [object[,]]) { throw "The report sheet is empty." }

$wanted = @("SubscriptionName", "SubscriptionId", "BA ID", "EIM", "EHC ID")
$col = @{}
for ($c = 1; $c -le $data.GetLength(1); $c++) {
    $h = (ConvertTo-CellString $data[1, $c]).Trim()
    if ($h -and -not $col.ContainsKey($h)) { $col[$h] = $c }
}
foreach ($h in $wanted) {
    if (-not $col.ContainsKey($h)) { throw "Header '$h' not found in row 1. Headers found: $($col.Keys -join ', ')" }
}

$rows = New-Object System.Collections.Generic.List[object]
for ($r = 2; $r -le $data.GetLength(0); $r++) {
    $v = @{}
    foreach ($h in $wanted) { $v[$h] = (ConvertTo-CellString $data[$r, $col[$h]]).Trim() }
    if (-not $v["SubscriptionName"] -and -not $v["SubscriptionId"] -and -not $v["BA ID"] -and -not $v["EHC ID"]) { continue }
    $rows.Add([pscustomobject]@{ Row = $r; V = $v })
}
if ($rows.Count -eq 0) { throw "No rows found in the report." }
Write-Log "Rows to process: $($rows.Count)"

# ---------- API helper ----------
function Hide-Key([string]$s) { if ($s -and $key) { return $s.Replace($key, "***") } return $s }

function Invoke-ApiCheck {
    # Returns @{ Ok; Http; Detail; Auth }
    param([string]$Ba, [string]$Ehc)

    $uri = $BaseUrl + $ApiPath + "?baid=" + [uri]::EscapeDataString($Ba) + "&ehcId=" + [uri]::EscapeDataString($Ehc) + "&code=" + $key

    for ($attempt = 1; $attempt -le $MaxRetries; $attempt++) {
        try {
            $resp = Invoke-WebRequest -UseBasicParsing -Uri $uri -Method GET -ContentType "application/json" -TimeoutSec $TimeoutSec -ErrorAction Stop
            $body = "$($resp.Content)".Trim()
            if ([int]$resp.StatusCode -eq 200 -and $body -and $body -notin @("null", "[]", "{}", '""')) {
                $info = ""
                try {
                    $j = $body | ConvertFrom-Json -ErrorAction Stop
                    if ($j.ehcStatus) { $info = "ehcStatus=$($j.ehcStatus)" }
                } catch {}
                return @{ Ok = $true; Http = [int]$resp.StatusCode; Detail = $info; Auth = $false }
            }
            return @{ Ok = $false; Http = [int]$resp.StatusCode; Detail = "empty response"; Auth = $false }
        }
        catch {
            $code = $null
            try { if ($_.Exception.Response) { $code = [int]$_.Exception.Response.StatusCode } } catch {}
            $msg = Hide-Key (($_.Exception.Message -split "`n")[0].Trim())

            if ($code -in 401, 403) { return @{ Ok = $false; Http = $code; Detail = "not authorised (check function key / access)"; Auth = $true } }
            if ($code -and $code -ne 429 -and $code -lt 500) {
                return @{ Ok = $false; Http = $code; Detail = $(if ($code -eq 404) { "Not Found (no data)" } else { $msg }); Auth = $false }
            }
            # no response (timeout / network), 429 or 5xx -> retry
            if ($attempt -eq $MaxRetries) {
                return @{ Ok = $false; Http = $code; Detail = $(if ($code) { "failed after $MaxRetries tries" } else { "no response: $msg" }); Auth = $false }
            }
            Start-Sleep -Seconds (2 * $attempt)
        }
    }
}

function Get-FailText($p) {
    if ($p.Http) { return "HTTP $($p.Http) - $($p.Detail)" }
    return "$($p.Detail)"
}

# ---------- Main loop ----------
# One record per OUTPUT row.  Status: GOOD | BAD | SKIPPED | MULTI
$records    = New-Object System.Collections.Generic.List[object]
$multiCells = New-Object System.Collections.Generic.List[object]   # detail for the log only
$i = 0
$authStreak = 0

function Add-Record {
    param([int]$Row, [hashtable]$V, [string]$Status, [string]$Notes)
    $records.Add([pscustomobject]@{ Row = $Row; V = $V; Status = $Status; Notes = $Notes })
}

foreach ($row in $rows) {
    $i++
    $v = $row.V
    $name = $v["SubscriptionName"]; $ba = $v["BA ID"]; $ehcRaw = $v["EHC ID"]
    Write-Progress -Activity "Calling BA/EHC API" -Status "$i of $($rows.Count): $name" -PercentComplete (($i / $rows.Count) * 100)
    $label = "$name  [BA=$ba  EHC=$ehcRaw]"

    $ehcList = @($ehcRaw -split '[,;]' | ForEach-Object { $_.Trim() } | Where-Object { $_ })
    $noEhc = ($ehcList.Count -eq 0)
    foreach ($p in $NoEhcValues) { if ($ehcRaw -ieq $p) { $noEhc = $true } }

    if ($noEhc) {
        $note = if ($ehcRaw) { "No EHC ID in report ('$ehcRaw') - not checked" } else { "No EHC ID in report - not checked" }
        Add-Record $row.Row $v "SKIPPED" $note
        Write-Log "SKIPPED   $label - $note" "Yellow"
        continue
    }
    if (-not $ba) {
        Add-Record $row.Row $v "BAD" "BA ID is blank - not sent to the API"
        Write-Log "BAD       $label - BA ID is blank (not sent)" "Red"
        continue
    }

    # Call the API once per EHC ID
    $perId = @()
    foreach ($e in $ehcList) {
        $res = Invoke-ApiCheck -Ba $ba -Ehc $e
        if ($res.Auth) { $authStreak++ } else { $authStreak = 0 }
        if ($authStreak -ge 3) {
            Write-Progress -Activity "Calling BA/EHC API" -Completed
            throw "Stopped: 3 calls in a row returned 401/403. Check the function key and your access. No output file was created."
        }
        $perId += [pscustomobject]@{ Ehc = $e; Ok = $res.Ok; Http = $res.Http; Detail = $res.Detail }
        if ($DelayMs -gt 0) { Start-Sleep -Milliseconds $DelayMs }
    }

    if ($ehcList.Count -eq 1) {
        $p = $perId[0]
        if ($p.Ok) {
            Add-Record $row.Row $v "GOOD" "OK"
            Write-Log "GOOD      $label$(if ($p.Detail) { ' - ' + $p.Detail })" "Green"
        }
        else {
            $note = Get-FailText $p
            Add-Record $row.Row $v "BAD" $note
            Write-Log "BAD       $label - $note" "Red"
        }
    }
    else {
        # Several EHC IDs in one cell -> one output row per ID, all BLUE
        $n = $ehcList.Count
        Write-Log "MULTI     $name  [BA=$ba] has $n comma-separated EHC IDs - split into $n rows, checked separately" "Cyan"
        foreach ($p in $perId) {
            $vv = @{}; foreach ($k in $v.Keys) { $vv[$k] = $v[$k] }
            $vv["EHC ID"] = $p.Ehc
            Add-Record $row.Row $vv "MULTI" "$n comma-separated EHC IDs found"
            if ($p.Ok) { Write-Log "            $($p.Ehc) -> data returned$(if ($p.Detail) { ' (' + $p.Detail + ')' })" "Green" }
            else       { Write-Log "            $($p.Ehc) -> NO DATA: $(Get-FailText $p)" "Red" }
        }
        $okCount = @($perId | Where-Object { $_.Ok }).Count
        if ($okCount -eq $n)      { Write-Log "            all $n IDs returned data" "Cyan" }
        elseif ($okCount -eq 0)   { Write-Log "            NONE of the $n IDs returned data - flag to the team" "Magenta" }
        else                      { Write-Log "            MIXED RESULT: $okCount of $n IDs returned data - flag to the team" "Magenta" }
        $multiCells.Add([pscustomobject]@{ Row = $row.Row; V = $v; PerId = $perId; OkCount = $okCount; Total = $n })
    }
}
Write-Progress -Activity "Calling BA/EHC API" -Completed

# ---------- Create the new Excel file ----------
$outHeaders = $wanted + @("Notes")
$saveError = $null
$xl = $null; $wbOut = $null; $wsOut = $null
try {
    $rowsN = $records.Count
    $colsN = $outHeaders.Count          # 6
    $arr   = New-Object 'object[,]' ($rowsN + 1), $colsN
    for ($c = 0; $c -lt $colsN; $c++) { $arr[0, $c] = $outHeaders[$c] }
    for ($r = 0; $r -lt $rowsN; $r++) {
        for ($c = 0; $c -lt $wanted.Count; $c++) { $arr[($r + 1), $c] = $records[$r].V[$wanted[$c]] }
        $arr[($r + 1), $wanted.Count] = $records[$r].Notes
    }

    # Visible + alerts ON so the sensitivity-label prompt can appear
    $xl = New-Object -ComObject Excel.Application
    $xl.Visible = $true
    $xl.DisplayAlerts = $true
    $wbOut = $xl.Workbooks.Add()
    $wsOut = $wbOut.Worksheets.Item(1)
    $wsOut.Name = "ApiValidation"

    $rng = $wsOut.Range($wsOut.Cells.Item(1, 1), $wsOut.Cells.Item($rowsN + 1, $colsN))
    $rng.NumberFormat = "@"          # keep every value as text
    $rng.Value2 = $arr

    $hdr = $wsOut.Range($wsOut.Cells.Item(1, 1), $wsOut.Cells.Item(1, $colsN))
    $hdr.Font.Bold = $true
    $hdr.Interior.Color = 15921906    # light grey

    # Excel colours are BGR: R + G*256 + B*65536
    $green  = 198 + (239 * 256) + (206 * 65536)    # light green
    $red    = 255 + (199 * 256) + (206 * 65536)    # light red
    $yellow = 255 + (235 * 256) + (156 * 65536)    # light yellow
    $blue   = 189 + (215 * 256) + (238 * 65536)    # light blue (multi-EHC rows)
    for ($r = 0; $r -lt $rowsN; $r++) {
        $color = switch ($records[$r].Status) { "GOOD" { $green } "BAD" { $red } "MULTI" { $blue } default { $yellow } }
        $xlRow = $r + 2
        $wsOut.Range($wsOut.Cells.Item($xlRow, 1), $wsOut.Cells.Item($xlRow, $colsN)).Interior.Color = $color
    }

    try { [void]$wsOut.Range($wsOut.Cells.Item(1, 1), $wsOut.Cells.Item(1, $colsN)).AutoFilter() } catch {}
    [void]$wsOut.Columns.AutoFit()

    # ----- Sensitivity label -----
    if ($LabelId -and $TenantId) {
        try {
            $sl   = $wbOut.SensitivityLabel
            $info = $sl.CreateLabelInfo()
            $info.LabelId          = $LabelId
            $info.LabelName        = $LabelName
            $info.SiteId           = $TenantId
            $info.AssignmentMethod = 1
            $info.IsEnabled        = $true
            $sl.SetLabel($info, $info)
            Write-Log "Sensitivity label '$LabelName' applied." "Green"
        }
        catch {
            Write-Log "WARNING: could not set the label automatically: $($_.Exception.Message)" "Yellow"
            Write-Log "If Excel shows a sensitivity prompt, choose '$LabelName' to continue." "Yellow"
        }
    }
    else {
        Write-Log "No label GUID given. If Excel shows a sensitivity prompt, choose '$LabelName' (the script waits)." "Yellow"
    }

    $wbOut.SaveAs($OutputPath, 51)    # 51 = .xlsx

    if (-not (Test-Path -LiteralPath $OutputPath)) {
        throw "Excel reported the save as successful, but no file exists at: $OutputPath (Excel's own workbook name is now '$($wbOut.FullName)'; a missing sensitivity label or security software may have blocked it)."
    }
    $saved = Get-Item -LiteralPath $OutputPath
    Write-Log "File saved and verified on disk: $($saved.FullName)  ($([math]::Round($saved.Length / 1KB)) KB)" "Green"
}
catch {
    $saveError = $_.Exception.Message
    Write-Log "ERROR: file could not be saved. $saveError" "Red"
}
finally {
    if ($wsOut) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wsOut) }
    if ($wbOut) { try { $wbOut.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wbOut) }
    if ($xl)    { try { $xl.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($xl) }
}

# ---------- Final summary ----------
$good    = @($records | Where-Object Status -eq "GOOD")
$bad     = @($records | Where-Object Status -eq "BAD")
$skipped = @($records | Where-Object Status -eq "SKIPPED")
$multiRows = @($records | Where-Object Status -eq "MULTI")
$checked = $good.Count + $bad.Count
$pct     = if ($checked) { [math]::Round(100 * $good.Count / $checked, 1) } else { 0 }
$multiOk   = ($multiCells | ForEach-Object { $_.OkCount } | Measure-Object -Sum).Sum
if (-not $multiOk) { $multiOk = 0 }
$multiFail = $multiRows.Count - $multiOk
$mixed     = @($multiCells | Where-Object { $_.OkCount -ne $_.Total })

Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  BA / EHC API VALIDATION SUMMARY"
Write-Plain ("=" * 78)
Write-Plain "  Run time                        : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Plain "  Report file                     : $fullReport"
Write-Plain "  Rows in report / rows in output : $($rows.Count) / $($records.Count)"
Write-Plain "  Single-EHC rows checked         : $checked"
Write-Plain "  GOOD  (data returned, green)    : $($good.Count)  ($pct% of checked)" "Green"
Write-Plain "  BAD   (no data / error, red)    : $($bad.Count)" $(if ($bad.Count) { "Red" } else { "Green" })
Write-Plain "  SKIPPED (no EHC ID, yellow)     : $($skipped.Count)" $(if ($skipped.Count) { "Yellow" } else { "Green" })
Write-Plain "  MULTI-EHC cells (blue)          : $($multiCells.Count)  (split into $($multiRows.Count) rows: $multiOk returned data, $multiFail did not)" $(if ($multiCells.Count) { "Cyan" } else { "Green" })
Write-Plain ("-" * 78)

if ($bad.Count -eq 0) {
    Write-Plain "  RESULT: every single-EHC row returned data from the API ($($good.Count) of $checked)." "Green"
}
else {
    Write-Plain "  RESULT: $($bad.Count) of $checked single-EHC row(s) returned NO DATA - details below." "Red"
    Write-Plain ""
    Write-Plain "  BAD ROWS ($($bad.Count)):" "Red"
    foreach ($b in $bad) {
        Write-Plain "  * $($b.V['SubscriptionName'])   (report row $($b.Row))"
        Write-Plain "    ID     : $($b.V['SubscriptionId'])"
        Write-Plain "    BA ID  : $($b.V['BA ID'])    EHC ID : $($b.V['EHC ID'])"
        Write-Plain "    Reason : $($b.Notes)"
    }
}

if ($multiCells.Count -gt 0) {
    Write-Plain ""
    Write-Plain "  MULTIPLE EHC IDs IN ONE CELL ($($multiCells.Count)) - result per ID (log only, to flag to the team):" "Cyan"
    foreach ($m in $multiCells) {
        Write-Plain "  * $($m.V['SubscriptionName'])   (report row $($m.Row))  [$($m.V['SubscriptionId'])]"
        Write-Plain "    BA ID  : $($m.V['BA ID'])"
        foreach ($p in $m.PerId) {
            if ($p.Ok) { Write-Plain "    $($p.Ehc)  ->  data returned" "Green" }
            else       { Write-Plain "    $($p.Ehc)  ->  NO DATA: $(Get-FailText $p)" "Red" }
        }
        if ($m.OkCount -ne $m.Total) { Write-Plain "    >> $($m.OkCount) of $($m.Total) returned data - FLAG TO TEAM" "Magenta" }
    }
}

if ($skipped.Count -gt 0) {
    Write-Plain ""
    Write-Plain "  SKIPPED - NO EHC ID, NOT SENT TO THE API ($($skipped.Count)) - for SRE:" "Yellow"
    foreach ($s in $skipped) {
        Write-Plain "  * $($s.V['SubscriptionName'])   (report row $($s.Row))  [$($s.V['SubscriptionId'])]"
        Write-Plain "    BA ID  : $($s.V['BA ID'])    EHC ID cell : $($s.V['EHC ID'])"
    }
}

Write-Plain ""
Write-Plain ("-" * 78)
if ($saveError) { Write-Plain "  OUTPUT FILE: NOT SAVED - $saveError" "Red" }
else {
    Write-Plain "  Output file : $OutputPath"
    Write-Plain "  (green = data returned, red = no data/error, yellow = not checked, blue = multi-EHC row)"
}
Write-Plain "  API: GET only. No Azure/portal calls. Report untouched. Key not stored or printed."
Write-Plain ("=" * 78)
