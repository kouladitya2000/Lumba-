# ================= CONFIG =================
$csvPath      = "C:\Users\45270978\Desktop\TagsValidator\SingleFinalEHC_20261020_030404.csv"   # read-only input
$auditFolder  = "C:\Users\45270978\Desktop\TagsValidator\Audit"                                # audit CSV goes here
$tagName      = "EHC ID"
$dryRun       = $true     # $true = read only, log what WOULD change. Set $false to really update.
$pauseBetween = 1         # seconds between subscriptions
# ==========================================

# ---------- Azure login check ----------
$ctx = Get-AzContext
if (-not $ctx) {
    Write-Host "Not logged in to Azure. Run Connect-AzAccount first, then re-run this script." -ForegroundColor Red
    return
}
Write-Host "Azure account : $($ctx.Account.Id)" -ForegroundColor Cyan

# ---------- Audit file ----------
if (-not (Test-Path $auditFolder)) { [void](New-Item -ItemType Directory -Path $auditFolder -Force) }
$stamp    = Get-Date -Format "yyyyMMdd_HHmmss"
$auditCsv = Join-Path $auditFolder "EHC_Tag_Audit_$stamp.csv"

# ---------- Load CSV ----------
$rows  = @(Import-Csv -Path $csvPath)
$total = $rows.Count
if ($total -eq 0) { Write-Host "CSV has no records." -ForegroundColor Red; return }

$headers = @($rows[0].PSObject.Properties | ForEach-Object { $_.Name })
foreach ($h in @('SubscriptionName','SubscriptionId','EHC ID')) {
    if ($headers -notcontains $h) {
        Write-Host "Column '$h' not found in CSV. Headers found: $($headers -join ', ')" -ForegroundColor Red
        return
    }
}

$modeText = "LIVE - tags WILL be updated"
if ($dryRun) { $modeText = "DRY RUN - nothing will be changed" }

Write-Host "`n############################################################" -ForegroundColor Cyan
Write-Host "MODE : $modeText" -ForegroundColor Yellow
Write-Host "RUN  : $total subscriptions from file" -ForegroundColor Cyan
Write-Host "AUDIT: $auditCsv" -ForegroundColor Cyan
Write-Host "############################################################`n" -ForegroundColor Cyan

# ---------- Helpers ----------
function Get-SubTags($subId) {
    $res = Get-AzTag -ResourceId "/subscriptions/$subId" -ErrorAction Stop
    $h = @{}
    if ($res -and $res.Properties -and $res.Properties.TagsProperty) {
        foreach ($k in $res.Properties.TagsProperty.Keys) { $h[$k] = $res.Properties.TagsProperty[$k] }
    }
    return $h
}

function Tags-ToText($h) {
    if ($h -eq $null -or $h.Count -eq 0) { return "{}" }
    $parts = @()
    foreach ($k in ($h.Keys | Sort-Object)) { $parts += "$k=$($h[$k])" }
    return ($parts -join "; ")
}

function Write-Audit($r) {
    $o = "" | Select-Object Timestamp, FileRow, SubscriptionName, SubscriptionId, Status, OldEHC, NewEHC, TagsBefore, TagsAfter, Message
    $o.Timestamp        = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
    $o.FileRow          = $r.Row
    $o.SubscriptionName = $r.SubName
    $o.SubscriptionId   = $r.SubId
    $o.Status           = $r.Status
    $o.OldEHC           = $r.OldEHC
    $o.NewEHC           = $r.NewEHC
    $o.TagsBefore       = $r.TagsBefore
    $o.TagsAfter        = $r.TagsAfter
    $o.Message          = $r.Message
    $o | Export-Csv -Path $auditCsv -Append -NoTypeInformation
}

function Show-Summary {
    $elapsed  = (Get-Date) - $startTime
    $updated  = @($results | Where-Object { $_.Status -eq "UPDATED" })
    $wouldUpd = @($results | Where-Object { $_.Status -eq "DRYRUN_WOULD_UPDATE" })
    $same     = @($results | Where-Object { $_.Status -eq "ALREADY_SET" })
    $noEhc    = @($results | Where-Object { $_.Status -eq "SKIPPED_NO_EHC" })
    $badId    = @($results | Where-Object { $_.Status -eq "SKIPPED_BAD_SUBID" })
    $failed   = @($results | Where-Object { $_.Status -eq "FAILED" })
    $otherChg = @($results | Where-Object { $_.Status -eq "UPDATED_BUT_OTHER_TAGS_CHANGED" })
    $cleaned  = @($results | Where-Object { $_.Cleaned })
    $dupes    = @($results | Where-Object { $_.Duplicate })

    Write-Host "`n==================== FINAL SUMMARY ====================" -ForegroundColor Cyan
    Write-Host "Mode                         : $modeText"
    Write-Host "Run time                     : $([int]$elapsed.TotalMinutes) min"
    Write-Host "Processed                    : $($results.Count) of $total"
    Write-Host "UPDATED                      : $($updated.Count)"  -ForegroundColor Green
    Write-Host "DRY RUN would update         : $($wouldUpd.Count)" -ForegroundColor Yellow
    Write-Host "Already had same EHC ID      : $($same.Count)"
    Write-Host "SKIPPED - no EHC in file     : $($noEhc.Count)"    -ForegroundColor Yellow
    Write-Host "SKIPPED - bad subscription id: $($badId.Count)"   -ForegroundColor Yellow
    Write-Host "FAILED                       : $($failed.Count)"   -ForegroundColor Red
    Write-Host "OTHER TAGS CHANGED (check!)  : $($otherChg.Count)" -ForegroundColor Red
    Write-Host "(Withdrawn) cleaned          : $($cleaned.Count)"
    Write-Host "Duplicate subscription ids   : $($dupes.Count)"

    Write-Host "`n---------- FAILED ----------" -ForegroundColor Red
    if ($failed.Count -eq 0) { Write-Host "None" -ForegroundColor Green }
    foreach ($r in $failed) {
        Write-Host "Row $($r.Row) | $($r.SubName) | $($r.SubId) | EHC=$($r.NewEHC)" -ForegroundColor Red
        Write-Host "   Error: $($r.Message)"
    }

    Write-Host "`n---------- OTHER TAGS CHANGED (investigate) ----------" -ForegroundColor Red
    if ($otherChg.Count -eq 0) { Write-Host "None" -ForegroundColor Green }
    foreach ($r in $otherChg) {
        Write-Host "Row $($r.Row) | $($r.SubName) | $($r.SubId)" -ForegroundColor Red
        Write-Host "   $($r.Message)"
    }

    Write-Host "`n---------- SKIPPED: NO EHC IN FILE (tags not touched) ----------" -ForegroundColor Yellow
    if ($noEhc.Count -eq 0) { Write-Host "None" }
    foreach ($r in $noEhc) { Write-Host "Row $($r.Row) | $($r.SubName) | $($r.SubId) | file value='$($r.OrigEHC)' | current tag='$($r.OldEHC)'" }

    Write-Host "`n---------- SKIPPED: BAD SUBSCRIPTION ID ----------" -ForegroundColor Yellow
    if ($badId.Count -eq 0) { Write-Host "None" }
    foreach ($r in $badId) { Write-Host "Row $($r.Row) | $($r.SubName) | id='$($r.SubId)'" }

    Write-Host "`n---------- (Withdrawn) CLEANED ----------" -ForegroundColor Yellow
    if ($cleaned.Count -eq 0) { Write-Host "None" }
    foreach ($r in $cleaned) { Write-Host "Row $($r.Row) | $($r.SubName) | '$($r.OrigEHC)' -> '$($r.NewEHC)'" }

    Write-Host "`n---------- ALREADY SET (no change needed) ----------" -ForegroundColor DarkGray
    if ($same.Count -eq 0) { Write-Host "None" }
    foreach ($r in $same) { Write-Host "Row $($r.Row) | $($r.SubName) | EHC ID already '$($r.NewEHC)'" }

    Write-Host "`n############################################################" -ForegroundColor Cyan
    Write-Host "AUDIT FILE: $auditCsv" -ForegroundColor Cyan
    Write-Host "############################################################" -ForegroundColor Cyan
}

# ================= MAIN LOOP =================
$results   = @()
$seenIds   = @{}
$startTime = Get-Date

try {
    for ($i = 1; $i -le $total; $i++) {
        $row = $rows[$i - 1]
        $tag = "[$i/$total]"

        $subName = ("" + $row.'SubscriptionName').Trim()
        $subId   = ("" + $row.'SubscriptionId').Trim()
        $origEhc = ("" + $row.'EHC ID').Trim()
        $newEhc  = ($origEhc -replace '\(.*?\)', '').Trim()
        $cleaned = ($newEhc -ne $origEhc)

        $rec = @{
            Row = $i; SubName = $subName; SubId = $subId; OrigEHC = $origEhc
            NewEHC = $newEhc; OldEHC = ""; Status = ""; Message = ""
            TagsBefore = ""; TagsAfter = ""; Cleaned = $cleaned; Duplicate = $false
        }

        # --- validate subscription id ---
        if ($subId -notmatch '^[0-9a-fA-F]{8}-([0-9a-fA-F]{4}-){3}[0-9a-fA-F]{12}$') {
            $rec.Status  = "SKIPPED_BAD_SUBID"
            $rec.Message = "Subscription id is not a valid GUID"
            Write-Host "$tag SKIP   $subName - bad subscription id '$subId'" -ForegroundColor Yellow
            $results += $rec; Write-Audit $rec
            continue
        }

        if ($seenIds.ContainsKey($subId)) {
            $rec.Duplicate = $true
            Write-Host "$tag WARN   duplicate subscription id (first seen at row $($seenIds[$subId]))" -ForegroundColor Yellow
        } else { $seenIds[$subId] = $i }

        if ($cleaned) { Write-Host "$tag INFO   cleaned '$origEhc' -> '$newEhc'" -ForegroundColor DarkGray }

        # --- read tags BEFORE ---
        $before = $null
        try { $before = Get-SubTags $subId }
        catch {
            $rec.Status  = "FAILED"
            $rec.Message = "Could not read tags: $($_.Exception.Message)"
            Write-Host "$tag FAILED $subName ($subId) - could not read tags: $($_.Exception.Message)" -ForegroundColor Red
            $results += $rec; Write-Audit $rec
            continue
        }

        $rec.TagsBefore = Tags-ToText $before
        $oldEhc = ""
        if ($before.ContainsKey($tagName)) { $oldEhc = "" + $before[$tagName] }
        $rec.OldEHC = $oldEhc

        # --- no EHC in file: log only ---
        if ($newEhc -eq "" -or $newEhc -match 'NO EHC FOUND' -or $newEhc -match '^API ERROR') {
            $rec.NewEHC    = ""
            $rec.Status    = "SKIPPED_NO_EHC"
            $rec.Message   = "No EHC ID in file ('$origEhc'). Tags not touched."
            $rec.TagsAfter = $rec.TagsBefore
            Write-Host "$tag SKIP   $subName ($subId) - NO EHC in file ('$origEhc'). Current tag value: '$oldEhc'. Not touched." -ForegroundColor Yellow
            $results += $rec; Write-Audit $rec
            continue
        }

        # --- already correct ---
        if ($oldEhc -ceq $newEhc) {
            $rec.Status    = "ALREADY_SET"
            $rec.Message   = "EHC ID already has the correct value"
            $rec.TagsAfter = $rec.TagsBefore
            Write-Host "$tag SAME   $subName - EHC ID already '$newEhc'" -ForegroundColor DarkGray
            $results += $rec; Write-Audit $rec
            continue
        }

        # --- dry run ---
        if ($dryRun) {
            $rec.Status    = "DRYRUN_WOULD_UPDATE"
            $rec.Message   = "Dry run - no change made"
            $rec.TagsAfter = $rec.TagsBefore
            Write-Host "$tag DRYRUN $subName ($subId) - EHC ID '$oldEhc' -> '$newEhc'" -ForegroundColor Yellow
            $results += $rec; Write-Audit $rec
            continue
        }

        # --- real update (Merge = only EHC ID is added/changed) ---
        try {
            [void](Update-AzTag -ResourceId "/subscriptions/$subId" -Tag @{ $tagName = $newEhc } -Operation Merge -ErrorAction Stop)
        }
        catch {
            $rec.Status    = "FAILED"
            $rec.Message   = "Update failed: $($_.Exception.Message)"
            $rec.TagsAfter = $rec.TagsBefore
            Write-Host "$tag FAILED $subName ($subId) - update error: $($_.Exception.Message)" -ForegroundColor Red
            $results += $rec; Write-Audit $rec
            continue
        }

        # --- read tags AFTER and verify nothing else changed ---
        $after = $null
        try { $after = Get-SubTags $subId } catch { $rec.Message = "Updated, but could not re-read tags: $($_.Exception.Message)" }

        if ($after -ne $null) {
            $rec.TagsAfter = Tags-ToText $after
            $problems = @()
            foreach ($k in $before.Keys) {
                if ($k -eq $tagName) { continue }
                if (-not $after.ContainsKey($k)) { $problems += "tag '$k' MISSING after update" }
                elseif ("" + $after[$k] -cne "" + $before[$k]) { $problems += "tag '$k' changed '$($before[$k])' -> '$($after[$k])'" }
            }
            foreach ($k in $after.Keys) {
                if ($k -eq $tagName) { continue }
                if (-not $before.ContainsKey($k)) { $problems += "tag '$k' NEW after update" }
            }
            $afterEhc = ""
            if ($after.ContainsKey($tagName)) { $afterEhc = "" + $after[$tagName] }

            if ($problems.Count -gt 0) {
                $rec.Status  = "UPDATED_BUT_OTHER_TAGS_CHANGED"
                $rec.Message = ($problems -join " | ")
                Write-Host "$tag ALERT  $subName ($subId) - other tags differ after update: $($rec.Message)" -ForegroundColor Red
            }
            elseif ($afterEhc -cne $newEhc) {
                $rec.Status  = "FAILED"
                $rec.Message = "Update call succeeded but EHC ID reads '$afterEhc' instead of '$newEhc'"
                Write-Host "$tag FAILED $subName ($subId) - $($rec.Message)" -ForegroundColor Red
            }
            else {
                $rec.Status  = "UPDATED"
                $rec.Message = "EHC ID updated, other tags verified unchanged"
                Write-Host "$tag OK     $subName ($subId) - EHC ID '$oldEhc' -> '$newEhc' | other tags verified unchanged" -ForegroundColor Green
            }
        }
        else {
            $rec.Status = "UPDATED"
            Write-Host "$tag OK*    $subName ($subId) - EHC ID '$oldEhc' -> '$newEhc' | $($rec.Message)" -ForegroundColor Yellow
        }

        $results += $rec; Write-Audit $rec
        Start-Sleep -Seconds $pauseBetween
    }
}
finally {
    Show-Summary
}
