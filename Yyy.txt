<#
.SYNOPSIS
  Validates the EHC ID column of the generated report against the LIVE "EHC ID" tag
  on each Azure subscription. Log check only: everything prints on screen.

  INPUT : the report .xlsx with columns SubscriptionName | SubscriptionId | ... | EHC ID
  CHECK : for every row, report "EHC ID" must equal the "EHC ID" tag on that subscription.
  RESULT: prints "100% MATCH" when every row agrees, otherwise a consolidated summary
          of every difference.

.READ-ONLY GUARANTEE
  - Azure : only reads (Get-AzSubscription, Get-AzTag). Nothing in Azure is changed.
  - Excel : the report is opened read-only and closed without saving.
  - Nothing is written anywhere: no report, no log file.

.PREREQUISITES
  - Microsoft Excel installed (used to read the report).
  - Install-Module Az.Accounts, Az.Resources -Scope CurrentUser
  - Connect-AzAccount

.COMPARISON RULES
  - Values are compared after trimming spaces, ignoring upper/lower case.
  - If a cell holds several values (e.g. "RITM1, RITM2"), the same values in a different
    order or with different spacing still count as a match, and are counted separately
    in the summary so you can see how many rows that applies to.
  - Report cell empty AND Azure tag missing/empty: counted as a match, but listed in the
    summary as a note (there is no EHC ID on either side).

.USAGE
  1. Set -ReportFile below (or pass it as a parameter).
  2. Run. Add -MismatchesOnly to hide the per-row OK lines.
     If the report has an active filter, hidden rows are INCLUDED by default;
     add -VisibleRowsOnly to check only the rows you can see.
#>

[CmdletBinding()]
param(
    # >>> SET THE REPORT PATH HERE <<<
    [string]$ReportFile = "C:\Users\45270978\Desktop\TagsValidator\Stage2SubscriptionTagsReportEHCValidateHelper_20261009_160619.xlsx",
    [string]$SheetName  = "",                      # blank = first sheet
    [string]$NameHeader = "SubscriptionName",
    [string]$IdHeader   = "SubscriptionId",
    [string]$EhcHeader  = "EHC ID",                # column header in the report
    [string]$TagName    = "EHC ID",                # tag name on the Azure subscription

    [switch]$MismatchesOnly,
    [switch]$VisibleRowsOnly,
    [int]$MaxRetries = 3
)

$ErrorActionPreference = "Stop"

function Write-Log {
    param([string]$Message, [string]$Color = "")
    $line = "[{0}] {1}" -f (Get-Date -Format "HH:mm:ss"), $Message
    if ($Color) { Write-Host $line -ForegroundColor $Color } else { Write-Host $line }
}
function Write-Plain {
    param([string]$Message = "", [string]$Color = "")
    if ($Color) { Write-Host $Message -ForegroundColor $Color } else { Write-Host $Message }
}

# ---------- Pre-flight ----------
if (-not (Get-Module -ListAvailable -Name Az.Resources)) {
    throw "Az.Resources module not found. Run: Install-Module Az.Accounts, Az.Resources -Scope CurrentUser"
}
if (-not (Get-AzContext)) { throw "Not signed in to Azure. Run Connect-AzAccount first." }
if (-not (Test-Path -LiteralPath $ReportFile)) {
    throw "Report file not found: $ReportFile  (set the correct path at the top of the script)"
}
$fullReport = (Resolve-Path -LiteralPath $ReportFile).Path

# ---------- Excel helpers (read-only) ----------
function ConvertTo-CellString($v) {
    if ($null -eq $v) { return "" }
    if ($v -is [double]) { return $v.ToString("0.##########", [Globalization.CultureInfo]::InvariantCulture) }
    return [string]$v
}

function Read-ReportSheet {
    param([string]$FullPath, [string]$Sheet)

    $excel = $null; $wb = $null; $ws = $null; $ur = $null
    $own = $false; $running = $null; $data = $null; $hidden = @{}

    try { $running = [Runtime.InteropServices.Marshal]::GetActiveObject("Excel.Application") } catch { $running = $null }
    if ($running) {
        foreach ($w in $running.Workbooks) {
            if ($w.FullName -ieq $FullPath) { $wb = $w; $excel = $running; break }
        }
    }

    try {
        if (-not $wb) {
            $excel = New-Object -ComObject Excel.Application
            $excel.Visible = $false
            $excel.DisplayAlerts = $false
            $own = $true
            $wb = $excel.Workbooks.Open($FullPath, 0, $true)      # UpdateLinks=0, ReadOnly=true
        }
        $ws = if ($Sheet) { $wb.Worksheets.Item($Sheet) } else { $wb.Worksheets.Item(1) }
        $ur = $ws.UsedRange
        $data = $ur.Value2
        $firstRow = [int]$ur.Row
        if ($data -is [object[,]]) {
            $n = $data.GetLength(0)
            for ($k = 2; $k -le $n; $k++) {
                if ([bool]$ws.Rows.Item($firstRow + $k - 1).Hidden) { $hidden[$k] = $true }
            }
        }
    }
    catch {
        throw "Could not read '$FullPath' (sheet '$Sheet'). $($_.Exception.Message)"
    }
    finally {
        if ($ur) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ur) }
        if ($ws) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ws) }
        if ($own) {
            if ($wb)    { try { $wb.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wb) }
            if ($excel) { try { $excel.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
        }
    }
    return @{ Data = $data; Hidden = $hidden }
}

# ---------- Read the report ----------
Write-Log "Starting EHC ID validation (Azure and Excel access are read-only)"
Write-Log "Report file : $fullReport"
$in   = Read-ReportSheet -FullPath $fullReport -Sheet $SheetName
$data = $in.Data
if ($data -isnot [object[,]]) { throw "The report sheet is empty." }

$col = @{}
for ($c = 1; $c -le $data.GetLength(1); $c++) {
    $h = (ConvertTo-CellString $data[1, $c]).Trim()
    if ($h -and -not $col.ContainsKey($h)) { $col[$h] = $c }
}
foreach ($h in @($NameHeader, $IdHeader, $EhcHeader)) {
    if (-not $col.ContainsKey($h)) { throw "Header '$h' not found in row 1. Headers found: $($col.Keys -join ', ')" }
}

$rows = New-Object System.Collections.Generic.List[object]
$hiddenCount = 0
for ($r = 2; $r -le $data.GetLength(0); $r++) {
    $n   = (ConvertTo-CellString $data[$r, $col[$NameHeader]]).Trim()
    $id  = (ConvertTo-CellString $data[$r, $col[$IdHeader]]).Trim()
    $ehc = (ConvertTo-CellString $data[$r, $col[$EhcHeader]]).Trim()
    if (-not $n -and -not $id -and -not $ehc) { continue }
    $isHidden = $in.Hidden.ContainsKey($r)
    if ($isHidden) { $hiddenCount++ }
    if ($isHidden -and $VisibleRowsOnly) { continue }
    $rows.Add([pscustomobject]@{ Row = $r; Name = $n; Id = $id; Ehc = $ehc })
}
if ($rows.Count -eq 0) { throw "No subscription rows found in the report." }

Write-Log "Rows to validate: $($rows.Count)"
if ($hiddenCount -gt 0) {
    if ($VisibleRowsOnly) { Write-Log "NOTE: $hiddenCount row(s) are hidden by a filter and were SKIPPED (-VisibleRowsOnly)." "Yellow" }
    else { Write-Log "NOTE: $hiddenCount row(s) are hidden by a filter and are INCLUDED. Use -VisibleRowsOnly to skip them." "Yellow" }
}

$dupes = $rows | Where-Object { $_.Id } | Group-Object { $_.Id.ToLower() } | Where-Object Count -gt 1
foreach ($d in $dupes) { Write-Log "WARNING: duplicate subscription ID in report: $($d.Name) ($($d.Count) rows)" "Yellow" }

# ---------- Azure subscription lookup (read-only) ----------
$subById = @{}; $subByName = @{}
try {
    foreach ($s in @(Get-AzSubscription -ErrorAction Stop)) {
        $subById[$s.Id.ToLower()] = $s
        $k = $s.Name.ToLower()
        if (-not $subByName.ContainsKey($k)) { $subByName[$k] = New-Object System.Collections.Generic.List[object] }
        $subByName[$k].Add($s)
    }
    Write-Log "Accessible subscriptions found in Azure: $($subById.Count)"
}
catch {
    Write-Log "WARNING: could not list subscriptions ($(($_.Exception.Message -split "`n")[0]))." "Yellow"
}

# ---------- Helpers ----------
function Get-SubscriptionTags {
    param([string]$SubId)
    for ($attempt = 1; $attempt -le $MaxRetries; $attempt++) {
        try {
            $t = Get-AzTag -ResourceId "/subscriptions/$SubId" -ErrorAction Stop
            return $t.Properties.TagsProperty
        }
        catch {
            $msg = $_.Exception.Message
            if ($msg -match "AuthorizationFailed|SubscriptionNotFound|InvalidAuthenticationToken|not found|Forbidden|does not have authorization") { throw }
            if ($attempt -eq $MaxRetries) { throw }
            Start-Sleep -Seconds (2 * $attempt)
        }
    }
}

function Get-TagValue {
    # Azure tag NAMES are case-insensitive
    param($Tags, [string]$Name)
    if (-not $Tags) { return @{ Exists = $false; Value = "" } }
    foreach ($k in $Tags.Keys) {
        if ($k -ieq $Name) { return @{ Exists = $true; Value = "$($Tags[$k])" } }
    }
    return @{ Exists = $false; Value = "" }
}

function Get-ValueSet {
    # "RITM2, RITM1" and "RITM1,RITM2" -> same normalised string
    param([string]$s)
    return (($s -split '[,;]' | ForEach-Object { $_.Trim().ToUpper() } | Where-Object { $_ } | Sort-Object -Unique) -join ',')
}

# ---------- Main loop ----------
# Statuses: MATCH | MATCH_FORMAT | BOTH_EMPTY | MISMATCH | TAG_NOT_SET | TAG_EMPTY | ERROR
$results = New-Object System.Collections.Generic.List[object]
$i = 0

foreach ($row in $rows) {
    $i++
    $name = $row.Name; $id = $row.Id; $reportVal = $row.Ehc
    $azureVal = ""; $err = $null; $status = $null; $note = ""

    Write-Progress -Activity "Validating EHC ID against Azure" -Status "$i of $($rows.Count): $name" -PercentComplete (($i / $rows.Count) * 100)

    # Resolve ID if missing
    if (-not $id) {
        $m = $subByName[$name.ToLower()]
        if (-not $m)            { $err = "Subscription ID is blank and the name was not found in accessible subscriptions." }
        elseif ($m.Count -gt 1) { $err = "Subscription name matches $($m.Count) subscriptions; the report needs the subscription ID." }
        else                    { $id = $m[0].Id }
    }
    else {
        $g = [guid]::Empty
        if (-not [guid]::TryParse($id, [ref]$g)) { $err = "'$id' is not a valid subscription ID." }
        elseif (-not $name -and $subById.ContainsKey($id.ToLower())) { $name = $subById[$id.ToLower()].Name }
    }

    if (-not $err) {
        try {
            $tags = Get-SubscriptionTags -SubId $id
            $tv   = Get-TagValue -Tags $tags -Name $TagName
            $azureVal = $tv.Value.Trim()

            if (-not $tv.Exists)           { $azureState = "NOT_SET" }
            elseif (-not $azureVal)        { $azureState = "EMPTY" }
            else                           { $azureState = "SET" }

            if (-not $reportVal -and $azureState -ne "SET") { $status = "BOTH_EMPTY" }
            elseif ($reportVal -and $azureState -eq "NOT_SET") { $status = "TAG_NOT_SET" }
            elseif ($reportVal -and $azureState -eq "EMPTY")   { $status = "TAG_EMPTY" }
            elseif ($reportVal -ieq $azureVal)                 { $status = "MATCH" }
            elseif ((Get-ValueSet $reportVal) -eq (Get-ValueSet $azureVal)) {
                $status = "MATCH_FORMAT"
                $note = "same values, different order/spacing"
            }
            else { $status = "MISMATCH" }
        }
        catch { $err = ($_.Exception.Message -split "`n")[0].Trim() }
    }
    if ($err) { $status = "ERROR" }

    $results.Add([pscustomobject]@{
        Row = $row.Row; Name = $name; Id = $id
        ReportValue = $reportVal; AzureValue = $azureVal
        Status = $status; Note = $note; Error = $err
    })

    $label = "$name ($id)"
    switch ($status) {
        "MATCH"        { if (-not $MismatchesOnly) { Write-Log "OK        $label" } }
        "MATCH_FORMAT" { if (-not $MismatchesOnly) { Write-Log "OK        $label - $note" } }
        "BOTH_EMPTY"   { Write-Log "NOTE      $label - no EHC ID in report and none in Azure" "Yellow" }
        "MISMATCH"     { Write-Log "MISMATCH  $label - report='$reportVal'  azure='$azureVal'" "Red" }
        "TAG_NOT_SET"  { Write-Log "MISMATCH  $label - report='$reportVal'  azure tag NOT SET" "Red" }
        "TAG_EMPTY"    { Write-Log "MISMATCH  $label - report='$reportVal'  azure tag EMPTY" "Red" }
        "ERROR"        { Write-Log "ERROR     $label - $err" "Red" }
    }
}
Write-Progress -Activity "Validating EHC ID against Azure" -Completed

# ---------- Final summary ----------
$total      = $results.Count
$exact      = @($results | Where-Object Status -eq "MATCH")
$fmt        = @($results | Where-Object Status -eq "MATCH_FORMAT")
$bothEmpty  = @($results | Where-Object Status -eq "BOTH_EMPTY")
$different  = @($results | Where-Object Status -eq "MISMATCH")
$notSet     = @($results | Where-Object Status -eq "TAG_NOT_SET")
$emptyTag   = @($results | Where-Object Status -eq "TAG_EMPTY")
$errors     = @($results | Where-Object Status -eq "ERROR")
$matched    = $exact.Count + $fmt.Count + $bothEmpty.Count
$problems   = $different.Count + $notSet.Count + $emptyTag.Count + $errors.Count
$pct        = if ($total) { [math]::Round(100 * $matched / $total, 1) } else { 0 }

Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  EHC ID VALIDATION SUMMARY   (report vs live Azure subscription tag '$TagName')"
Write-Plain ("=" * 78)
Write-Plain "  Run time                       : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Plain "  Report file                    : $fullReport"
Write-Plain "  Subscriptions checked          : $total"
Write-Plain "  Matching Azure                 : $matched  ($pct%)" $(if ($matched -eq $total) { "Green" } else { "Yellow" })
Write-Plain "  Different value in Azure       : $($different.Count)" $(if ($different.Count) { "Red" } else { "Green" })
Write-Plain "  Tag not set in Azure           : $($notSet.Count)"    $(if ($notSet.Count)    { "Red" } else { "Green" })
Write-Plain "  Tag empty in Azure             : $($emptyTag.Count)"  $(if ($emptyTag.Count)  { "Red" } else { "Green" })
Write-Plain "  Could not be read (errors)     : $($errors.Count)"    $(if ($errors.Count)    { "Red" } else { "Green" })
if ($hiddenCount -gt 0) {
    Write-Plain "  Rows hidden by report filter   : $hiddenCount ($(if ($VisibleRowsOnly) { 'skipped' } else { 'included' }))"
}
Write-Plain ("-" * 78)

if ($problems -eq 0) {
    Write-Plain "  RESULT: 100% MATCH - the EHC ID in the report equals the EHC ID tag on all $total subscriptions." "Green"
}
else {
    Write-Plain "  RESULT: NOT 100% - $problems of $total subscription(s) need attention ($pct% match)." "Red"

    if ($different.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  DIFFERENT VALUE ($($different.Count)):" "Red"
        foreach ($d in $different) {
            Write-Plain "  * $($d.Name)   (report row $($d.Row))"
            Write-Plain "    ID     : $($d.Id)"
            Write-Plain "    Report : $($d.ReportValue)"
            Write-Plain "    Azure  : $($d.AzureValue)"
        }
    }
    if ($notSet.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  TAG NOT SET IN AZURE ($($notSet.Count)):" "Red"
        foreach ($d in $notSet) {
            Write-Plain "  * $($d.Name)   (report row $($d.Row))  [$($d.Id)]"
            Write-Plain "    Report : $($d.ReportValue)"
        }
    }
    if ($emptyTag.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  TAG EMPTY IN AZURE ($($emptyTag.Count)):" "Red"
        foreach ($d in $emptyTag) {
            Write-Plain "  * $($d.Name)   (report row $($d.Row))  [$($d.Id)]"
            Write-Plain "    Report : $($d.ReportValue)"
        }
    }
    if ($errors.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  COULD NOT BE READ ($($errors.Count)) - check ID, access or tenant:" "Red"
        foreach ($d in $errors) {
            Write-Plain "  * $($d.Name)   (report row $($d.Row))  [$($d.Id)]"
            Write-Plain "    $($d.Error)"
        }
    }
}

if ($fmt.Count -gt 0 -or $bothEmpty.Count -gt 0) {
    Write-Plain ""
    Write-Plain "  NOTES (counted as matching):" "Yellow"
    if ($fmt.Count -gt 0) {
        Write-Plain "  * $($fmt.Count) row(s) hold several values in a different order/spacing than Azure:"
        foreach ($d in $fmt) { Write-Plain "      $($d.Name): report='$($d.ReportValue)'  azure='$($d.AzureValue)'" }
    }
    if ($bothEmpty.Count -gt 0) {
        Write-Plain "  * $($bothEmpty.Count) row(s) have no EHC ID in the report and none in Azure:"
        foreach ($d in $bothEmpty) { Write-Plain "      $($d.Name)  [$($d.Id)]" }
    }
}

Write-Plain ""
Write-Plain ("-" * 78)
Write-Plain "  Read-only: nothing changed in Azure, no files written or modified."
Write-Plain ("=" * 78)
