# ---- Paths (edit these) ----
$file1 = "C:\Path\baeimcorrect.csv"
$file2 = "C:\Path\Stage2SubscriptionTagsReportEHCValidateHelper_20261009_160619_ApiValidated.xlsx"
$sheetName = "Stage2SubscriptionTagsReportEHC"

# Removes spaces, tabs, line breaks, non-breaking and zero-width spaces anywhere in the value
function Norm($v) {
    if ($null -eq $v) { return "" }
    ([string]$v -replace '[\s\u00A0\u200B\uFEFF]', '').ToUpper()
}

# ---- Load reference (file 1): BA -> hashtable of valid EIMs ----
$ref = @{}
foreach ($r in (Import-Csv $file1 -Encoding UTF8)) {
    $ba  = Norm $r.'Number'
    $eim = Norm $r.'EIM Application Instance ID'
    if (-not $ref.ContainsKey($ba)) { $ref[$ba] = @{} }
    $ref[$ba][$eim] = $true
}

# ---- Open file 2 via Excel COM (read-only, never saved) ----
$xl = New-Object -ComObject Excel.Application
$xl.Visible = $false
$xl.DisplayAlerts = $false
$wb = $null
$failures = @()
$total = 0

try {
    $wb = $xl.Workbooks.Open($file2, 0, $true)   # UpdateLinks=0, ReadOnly=true
    $ws = $wb.Worksheets.Item($sheetName)

    # Find headers by name so column order doesn't matter
    $lastCol = $ws.UsedRange.Columns.Count
    $col = @{}
    for ($c = 1; $c -le $lastCol; $c++) {
        $h = ([string]$ws.Cells.Item(1, $c).Text).Trim()
        if ($h -and -not $col.ContainsKey($h)) { $col[$h] = $c }
    }
    foreach ($need in 'SubscriptionName','BA ID','EIM') {
        if (-not $col.ContainsKey($need)) { throw "Column '$need' not found in header row" }
    }

    $lastRow = $ws.UsedRange.Rows.Count
    for ($row = 2; $row -le $lastRow; $row++) {
        $sub = ([string]$ws.Cells.Item($row, $col['SubscriptionName']).Text).Trim()
        $ba  = Norm $ws.Cells.Item($row, $col['BA ID']).Text
        $eim = Norm $ws.Cells.Item($row, $col['EIM']).Text

        if (-not $sub -and -not $ba -and -not $eim) { continue }   # skip fully blank rows
        $total++

        if (-not $ref.ContainsKey($ba)) {
            $reason = "BA ID not found in file 1"; $expected = ""
        }
        elseif (-not $ref[$ba].ContainsKey($eim)) {
            $reason = "EIM mismatch"; $expected = ($ref[$ba].Keys -join ", ")
        }
        else { continue }

        $failures += [pscustomobject]@{
            ExcelRow     = $row
            Subscription = $sub
            BA_ID        = $ba
            EIM_in_File2 = $eim
            EIM_Expected = $expected
            Reason       = $reason
        }
    }
}
finally {
    if ($wb) { $wb.Close($false) }
    $xl.Quit()
    [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($xl)
}

# ---- Summary ----
$bad = $failures.Count
$pct = if ($total) { [math]::Round((($total - $bad) / $total) * 100, 2) } else { 0 }

Write-Host "`n===== COMPARISON SUMMARY =====" -ForegroundColor Cyan
Write-Host "Rows checked : $total"
Write-Host "Matched      : $($total - $bad)"
Write-Host "Failed       : $bad"

if ($bad -eq 0) {
    Write-Host "Result       : 100% MATCH" -ForegroundColor Green
} else {
    Write-Host "Result       : $pct% match" -ForegroundColor Red
    $failures | Format-Table -AutoSize
}
