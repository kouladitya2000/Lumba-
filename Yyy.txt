# Requires Az.Accounts and Az.Resources, and Connect-AzAccount first

# Get Decommissioned MG with all nested children
$mg = Get-AzManagementGroup -GroupName 'Decommissioned' -Expand -Recurse

# Walk the tree and collect every subscription ID under it
function Get-MgSubscriptionIds($group) {
    foreach ($child in $group.Children) {
        if ($child.Type -eq '/subscriptions') {
            $child.Name                      # Name = subscription ID
        }
        elseif ($child.Type -like '*managementGroups') {
            Get-MgSubscriptionIds $child     # recurse into nested MGs
        }
    }
}

$excludedIds = @(Get-MgSubscriptionIds $mg)

# Enabled subscriptions, minus the Decommissioned ones
$enabledSubs = Get-AzSubscription |
    Where-Object { $_.State -eq 'Enabled' -and $_.Id -notin $excludedIds }

"Enabled subscriptions (excluding Decommissioned): $($enabledSubs.Count)"
