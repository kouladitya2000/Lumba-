# ---- Paths (edit these) ----
$file1 = "C:\Path\baeimcorrect.csv"
$file2 = "C:\Path\report.csv"

# Removes spaces, tabs, line breaks, non-breaking and zero-width spaces anywhere in the value
function Norm($v) {
    if ($null -eq $v) { return "" }
    ([string]$v -replace '[\s\u00A0\u200B\uFEFF]', '').ToUpper()
}

# ---- Load file 1 (correct values): BA -> valid EIMs ----
$ref = @{}
foreach ($r in (Import-Csv $file1 -Encoding UTF8)) {
    $ba  = Norm $r.'Number'
    $eim = Norm $r.'EIM Application Instance ID'
    if (-not $ref.ContainsKey($ba)) { $ref[$ba] = @{} }
    $ref[$ba][$eim] = $true
}
if ($ref.Count -eq 0) { throw "File 1 loaded 0 rows - check path and headers (Number / EIM Application Instance ID)" }

# ---- Load file 2 ----
$rows = @(Import-Csv $file2 -Encoding UTF8)
if ($rows.Count -eq 0) { throw "File 2 loaded 0 rows - check the path" }

$cols = $rows[0].PSObject.Properties.Name
foreach ($need in 'SubscriptionName','BA ID','EIM') {
    if ($cols -notcontains $need) { throw "Column '$need' not found in file 2. Headers seen: $($cols -join ', ')" }
}

Write-Host "File 1 loaded: $($ref.Count) unique BA IDs" -ForegroundColor DarkGray
Write-Host "File 2 loaded: $($rows.Count) rows" -ForegroundColor DarkGray

# ---- Compare ----
$failures = @()
$total = 0
$rowNum = 1   # header = row 1

foreach ($r in $rows) {
    $rowNum++
    $sub = ([string]$r.'SubscriptionName').Trim()
    $ba  = Norm $r.'BA ID'
    $eim = Norm $r.'EIM'

    if (-not $sub -and -not $ba -and -not $eim) { continue }   # skip blank rows
    $total++

    if (-not $ref.ContainsKey($ba)) {
        $reason = "BA ID not found in file 1"; $expected = ""
    }
    elseif (-not $ref[$ba].ContainsKey($eim)) {
        $reason = "EIM mismatch"; $expected = ($ref[$ba].Keys -join ", ")
    }
    else { continue }

    $failures += [pscustomobject]@{
        CsvRow       = $rowNum
        Subscription = $sub
        BA_ID        = $ba
        EIM_in_File2 = $eim
        EIM_Expected = $expected
        Reason       = $reason
    }
}

# ---- Summary ----
$bad = $failures.Count
$pct = [math]::Round((($total - $bad) / $total) * 100, 2)

Write-Host "`n===== COMPARISON SUMMARY =====" -ForegroundColor Cyan
Write-Host "Rows checked : $total"
Write-Host "Matched      : $($total - $bad)"
Write-Host "Failed       : $bad"

if ($bad -eq 0) {
    Write-Host "Result       : 100% MATCH" -ForegroundColor Green
} else {
    Write-Host "Result       : $pct% match" -ForegroundColor Red
    $failures | Format-Table -AutoSize
}
