<#
.SYNOPSIS
  Checks that the BA ID + EIM pairs in your subscription-tags workbook
  (Stage1Analysis-TagsMapping.xlsx: column C = BA ID, column D = EIM)
  match the official mapping in the CMDB workbook
  (column A "Number" = BA ID, column F "EIM Application Instance ID" = EIM).

  A subscription is a MATCH when its exact (BA ID, EIM) pair exists as a row
  in the CMDB file.

  READ-ONLY and FILE-FREE:
    - Both workbooks are opened READ-ONLY through Excel (nothing saved or edited).
    - It does NOT connect to Azure at all.
    - Writes NOTHING to disk. All logs and the final summary print on screen.

.USAGE
  1. Set the two paths in the param block below (or pass them as parameters).
  2. Run the script (F5, or  .\Verify-BAID-EIM-Mapping.ps1 ).
  Each workbook may be open or closed in Excel; both work.
#>

[CmdletBinding()]
param(
    # >>> SET YOUR TWO FILE PATHS HERE <<<
    [string]$TagsExcelPath = "C:\Path\To\Stage1Analysis-TagsMapping.xlsx",
    [string]$CmdbExcelPath = "C:\Path\To\cmdb_ci_business_app.xlsx",

    # Sheet names
    [string]$TagsSheetName = "Sheet1",
    [string]$CmdbSheetName = "cmdb_ci_business_app",

    # Header names (row 1)
    [string]$TagsNameHeader = "SubscriptionName",
    [string]$TagsIdHeader   = "SubscriptionId",
    [string]$TagsBAHeader   = "BA ID",
    [string]$TagsEimHeader  = "EIM",
    [string]$CmdbBAHeader   = "Number",
    [string]$CmdbEimHeader  = "EIM Application Instance ID"
)

$ErrorActionPreference = "Stop"

function Write-Log {
    param([string]$Message, [string]$Color = "")
    $line = "[{0}] {1}" -f (Get-Date -Format "HH:mm:ss"), $Message
    if ($Color) { Write-Host $line -ForegroundColor $Color } else { Write-Host $line }
}
function Write-Plain {
    param([string]$Message = "", [string]$Color = "")
    if ($Color) { Write-Host $Message -ForegroundColor $Color } else { Write-Host $Message }
}

# ---------- Excel helpers (read-only) ----------
function Get-SheetData {
    param([string]$Path, [string]$Sheet)

    if (-not (Test-Path -LiteralPath $Path)) { throw "File not found: $Path  (set the correct path at the top of the script)" }
    $full = (Resolve-Path -LiteralPath $Path).Path

    $excel = $null; $wb = $null; $ws = $null; $own = $false; $running = $null

    # If the file is already open in a running Excel, read it from there
    try { $running = [Runtime.InteropServices.Marshal]::GetActiveObject("Excel.Application") } catch { $running = $null }
    if ($running) {
        foreach ($w in $running.Workbooks) {
            if ($w.FullName -ieq $full) { $wb = $w; $excel = $running; break }
        }
    }

    try {
        if (-not $wb) {
            # Otherwise open READ-ONLY in a separate, hidden Excel instance
            $excel = New-Object -ComObject Excel.Application
            $excel.Visible = $false
            $excel.DisplayAlerts = $false
            $own = $true
            $wb = $excel.Workbooks.Open($full, 0, $true)     # UpdateLinks=0, ReadOnly=true
        }
        $ws   = $wb.Worksheets.Item($Sheet)
        $data = $ws.UsedRange.Value2
    }
    catch {
        throw "Could not read sheet '$Sheet' from '$full'. If the file is open, make sure no cell is in edit mode. $($_.Exception.Message)"
    }
    finally {
        if ($ws) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ws) }
        if ($own) {
            if ($wb)    { try { $wb.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wb) }
            if ($excel) { try { $excel.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
        }
    }

    if ($data -isnot [object[,]]) { return $null }     # empty sheet
    return ,$data
}

function ConvertTo-CellString($v) {
    if ($null -eq $v) { return "" }
    if ($v -is [double]) { return $v.ToString("0.##########", [Globalization.CultureInfo]::InvariantCulture) }
    return [string]$v
}

function Get-HeaderMap {
    param($Data, [string]$Label, [string[]]$Required)
    $map = @{}
    for ($c = 1; $c -le $Data.GetLength(1); $c++) {
        $h = (ConvertTo-CellString $Data[1, $c]).Trim()
        if ($h -and -not $map.ContainsKey($h)) { $map[$h] = $c }
    }
    foreach ($h in $Required) {
        if (-not $map.ContainsKey($h)) {
            throw "[$Label] Header '$h' not found in row 1. Headers found: $($map.Keys -join ', ')"
        }
    }
    return $map
}

function Format-Short {
    param($Items, [int]$Max = 5)
    $arr = @($Items | Sort-Object)
    if ($arr.Count -eq 0) { return "none" }
    if ($arr.Count -le $Max) { return ($arr -join ", ") }
    return (($arr[0..($Max - 1)] -join ", ") + " (+$($arr.Count - $Max) more)")
}

# ---------- Load CMDB mapping ----------
Write-Log "Starting BA ID / EIM mapping verification (read-only, no files written, no Azure access)"
Write-Log "Reading CMDB mapping: $CmdbExcelPath"
$cm     = Get-SheetData -Path $CmdbExcelPath -Sheet $CmdbSheetName
$cmCols = $null
if ($cm) { $cmCols = Get-HeaderMap -Data $cm -Label "CMDB" -Required @($CmdbBAHeader, $CmdbEimHeader) }

$exactPairs = New-Object 'System.Collections.Generic.HashSet[string]'
$normPairs  = New-Object 'System.Collections.Generic.HashSet[string]'
$baToEims   = New-Object 'System.Collections.Generic.Dictionary[string,System.Collections.Generic.HashSet[string]]'
$eimToBas   = New-Object 'System.Collections.Generic.Dictionary[string,System.Collections.Generic.HashSet[string]]'

$cmRows = 0
$cmLast = if ($cm) { $cm.GetLength(0) } else { 0 }
for ($r = 2; $r -le $cmLast; $r++) {
    $ba  = ConvertTo-CellString $cm[$r, $cmCols[$CmdbBAHeader]]
    $eim = ConvertTo-CellString $cm[$r, $cmCols[$CmdbEimHeader]]
    if (-not $ba.Trim()) { continue }
    $cmRows++

    [void]$exactPairs.Add("$ba|$eim")
    $nb = $ba.Trim().ToLowerInvariant()
    $ne = $eim.Trim().ToLowerInvariant()
    [void]$normPairs.Add("$nb|$ne")

    if (-not $baToEims.ContainsKey($nb)) { $baToEims[$nb] = New-Object 'System.Collections.Generic.HashSet[string]' }
    if ($ne) {
        [void]$baToEims[$nb].Add($ne)
        if (-not $eimToBas.ContainsKey($ne)) { $eimToBas[$ne] = New-Object 'System.Collections.Generic.HashSet[string]' }
        [void]$eimToBas[$ne].Add($ba.Trim())
    }
}
$multiEim = @($baToEims.GetEnumerator() | Where-Object { $_.Value.Count -gt 1 }).Count
Write-Log "CMDB rows loaded: $cmRows  |  distinct BA IDs: $($baToEims.Count)  |  BA IDs with more than one EIM: $multiEim"
if ($cmRows -eq 0) {
    Write-Log "WARNING: the CMDB sheet has NO mapping rows. Every subscription will be reported as NO_MAPPING_IN_CMDB." "Yellow"
}

# ---------- Load subscription-tags workbook ----------
Write-Log "Reading subscription tags workbook: $TagsExcelPath"
$tg     = Get-SheetData -Path $TagsExcelPath -Sheet $TagsSheetName
if (-not $tg) { throw "The subscription tags sheet '$TagsSheetName' is empty." }
$tgCols = Get-HeaderMap -Data $tg -Label "TAGS" -Required @($TagsNameHeader, $TagsIdHeader, $TagsBAHeader, $TagsEimHeader)

$rows = for ($r = 2; $r -le $tg.GetLength(0); $r++) {
    $name = (ConvertTo-CellString $tg[$r, $tgCols[$TagsNameHeader]]).Trim()
    $id   = (ConvertTo-CellString $tg[$r, $tgCols[$TagsIdHeader]]).Trim()
    if (-not $name -and -not $id) { continue }
    [pscustomobject]@{
        Row = $r; SubscriptionName = $name; SubscriptionId = $id
        BA  = ConvertTo-CellString $tg[$r, $tgCols[$TagsBAHeader]]
        EIM = ConvertTo-CellString $tg[$r, $tgCols[$TagsEimHeader]]
    }
}
$rows = @($rows)
Write-Log "Subscriptions to check: $($rows.Count)"

# ---------- Main loop ----------
$results = New-Object System.Collections.Generic.List[object]
$i = 0

foreach ($row in $rows) {
    $i++
    Write-Progress -Activity "Checking BA ID / EIM against CMDB" -Status "$i of $($rows.Count): $($row.SubscriptionName)" -PercentComplete (($i / $rows.Count) * 100)

    $ba = $row.BA; $eim = $row.EIM
    $nb = $ba.Trim().ToLowerInvariant()
    $ne = $eim.Trim().ToLowerInvariant()

    if ($exactPairs.Contains("$ba|$eim")) {
        $status = "MATCH"; $detail = ""
    }
    elseif ($normPairs.Contains("$nb|$ne")) {
        $status = "FORMAT_DIFFERENCE"
        $detail = "Pair matches CMDB except for extra spaces or letter case."
    }
    elseif ($nb -eq "" -or $nb -eq "nf") {
        if ($ne -and $eimToBas.ContainsKey($ne)) {
            $status = "BA_ID_NF_OR_BLANK"
            $detail = "No BA ID in sheet. In CMDB, EIM $eim belongs to BA ID(s): $(Format-Short $eimToBas[$ne])"
        }
        else {
            $status = "NO_MAPPING_IN_CMDB"
            $detail = "No BA ID in sheet, and EIM '$eim' is not found anywhere in the CMDB."
        }
    }
    elseif (-not $baToEims.ContainsKey($nb)) {
        if ($ne -and $eimToBas.ContainsKey($ne)) {
            $status = "BA_ID_NOT_IN_CMDB"
            $detail = "BA ID not in CMDB. In CMDB, EIM $eim belongs to BA ID(s): $(Format-Short $eimToBas[$ne])"
        }
        else {
            $status = "NO_MAPPING_IN_CMDB"
            $detail = "Neither BA ID '$ba' nor EIM '$eim' exists anywhere in the CMDB."
        }
    }
    elseif ($baToEims[$nb].Count -eq 0) {
        $status = "BA_ID_IN_CMDB_WITHOUT_EIM"
        $detail = "BA ID $ba exists in CMDB but has no EIM filled in."
        if ($ne -and $eimToBas.ContainsKey($ne)) { $detail += "  |  EIM $eim is mapped in CMDB to BA ID(s): $(Format-Short $eimToBas[$ne])" }
    }
    elseif ($ne -eq "") {
        $status = "EIM_BLANK_IN_SHEET"
        $detail = "EIM is blank. CMDB EIM(s) for $ba : $(Format-Short $baToEims[$nb])"
    }
    else {
        $status = "EIM_MISMATCH"
        $detail = "CMDB EIM(s) for $ba : $(Format-Short $baToEims[$nb])"
        if ($eimToBas.ContainsKey($ne)) { $detail += "  |  EIM $eim is mapped in CMDB to BA ID(s): $(Format-Short $eimToBas[$ne])" }
        else { $detail += "  |  EIM $eim is not in CMDB at all." }
    }

    $results.Add([pscustomobject]@{
        Row = $row.Row; SubscriptionName = $row.SubscriptionName; SubscriptionId = $row.SubscriptionId
        BA = $ba; EIM = $eim; Status = $status; Detail = $detail
    })

    if ($status -eq "MATCH") { Write-Log "OK        $($row.SubscriptionName)  [$ba / $eim]" }
    else { Write-Log "MISMATCH  $($row.SubscriptionName)  [$ba / $eim]  - $status" "Yellow" }
}
Write-Progress -Activity "Checking BA ID / EIM against CMDB" -Completed

# ---------- Final summary (all together) ----------
$total   = $results.Count
$matched = @($results | Where-Object Status -eq "MATCH").Count
$bad     = @($results | Where-Object Status -ne "MATCH")
$pct     = if ($total) { [math]::Round(($matched / $total) * 100, 2) } else { 0 }

Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  BA ID / EIM MAPPING SUMMARY   (subscription tags file vs CMDB file)"
Write-Plain ("=" * 78)
Write-Plain "  Run time              : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Plain "  Subscriptions checked : $total"
Write-Plain "  Matched CMDB mapping  : $matched" "Green"
Write-Plain "  Not matching          : $($bad.Count)" $(if ($bad.Count) { "Yellow" } else { "Green" })
Write-Plain "  Match rate            : $pct%"
Write-Plain ("-" * 78)

if ($matched -eq $total -and $total -gt 0) {
    Write-Plain "  RESULT: 100% MATCH - every BA ID + EIM pair in the tags file exists in the CMDB mapping." "Green"
}
else {
    Write-Plain "  RESULT: NOT A 100% MATCH - $($bad.Count) subscription(s) need attention." "Red"

    Write-Plain ""
    Write-Plain "  Issue breakdown:"
    $bad | Group-Object Status | Sort-Object Name | ForEach-Object {
        Write-Plain ("    {0,-26} {1}" -f $_.Name, $_.Count)
    }

    $pairCount = @($bad | Group-Object { "$($_.BA)|$($_.EIM)" }).Count
    Write-Plain "    (distinct BA ID / EIM pairs involved: $pairCount)"

    Write-Plain ""
    Write-Plain "  NON-MATCHING SUBSCRIPTIONS ($($bad.Count)):" "Yellow"
    foreach ($g in ($bad | Group-Object Status | Sort-Object Name)) {
        Write-Plain ""
        Write-Plain "  --- $($g.Name) ($($g.Count)) ---" "Yellow"
        foreach ($b in $g.Group) {
            Write-Plain ""
            Write-Plain "  * $($b.SubscriptionName)   (Excel row $($b.Row))"
            Write-Plain "    ID    : $($b.SubscriptionId)"
            Write-Plain "    Tags file : BA ID='$($b.BA)'  EIM='$($b.EIM)'"
            Write-Plain "    Detail    : $($b.Detail)"
        }
    }
}
Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  No files were created or modified. Both Excel workbooks left untouched."
Write-Plain "  No Azure connection was used."
Write-Plain ("=" * 78)
