AzureActivity
| where TimeGenerated > ago(365d)
| where SubscriptionId == "<your-sub-id>"
| where OperationNameValue has "TAGS"
| where ResourceId !has "/resourceGroups/"
| extend Props = parse_json(Properties)
| project TimeGenerated, Caller, CallerIpAddress, OperationNameValue,
          ActivityStatusValue, ResourceId,
          RequestBody = Props.requestbody, ResponseBody = Props.responseBody
| order by TimeGenerated desc
