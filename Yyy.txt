# ================= Settings (edit these) =================
$inputCsv = "$env:USERPROFILE\Downloads\SingleFinalEHC_20261020_030404.csv"
$tagName  = "BA ID"
$tenantId = ""     # optional: your tenant id if you have several tenants
# =========================================================

if (-not (Get-Module -ListAvailable -Name Az.Accounts)) {
    throw "Az module not found. Run: Install-Module Az.Accounts -Scope CurrentUser"
}
if (-not (Test-Path $inputCsv)) { throw "Input file not found: $inputCsv" }

if (-not (Get-AzContext)) {
    if ($tenantId) { Connect-AzAccount -Tenant $tenantId | Out-Null } else { Connect-AzAccount | Out-Null }
}

$rows = @(Import-Csv $inputCsv -Encoding UTF8)
if ($rows.Count -eq 0) { throw "Input CSV loaded 0 rows" }
$cols = $rows[0].PSObject.Properties.Name
foreach ($need in 'SubscriptionName','SubscriptionId','BA ID') {
    if ($cols -notcontains $need) { throw "Column '$need' not found. Headers: $($cols -join ', ')" }
}

function Norm($v) { (([string]$v) -replace '\s', '').ToUpper() }
function NormKey($v) { (([string]$v) -replace '[\s_\-]', '').ToUpper() }

$cache = @{}
function Get-SubInfo($subId) {
    if ($cache.ContainsKey($subId)) { return $cache[$subId] }
    $info = @{ Found = $false; Error = ""; Name = ""; Tags = $null }
    try {
        $resp = Invoke-AzRestMethod -Method GET -Path "/subscriptions/$($subId)?api-version=2022-12-01"
        if ($resp.StatusCode -eq 200) {
            $j = $resp.Content | ConvertFrom-Json
            $info.Found = $true
            $info.Name  = [string]$j.displayName
            $info.Tags  = $j.tags
        }
        else { $info.Error = "HTTP $($resp.StatusCode)" }
    }
    catch { $info.Error = $_.Exception.Message }
    $cache[$subId] = $info
    return $info
}

$problems = @()
$ok = 0; $n = 0; $checked = 0

foreach ($r in $rows) {
    $n++
    $sub   = ([string]$r.'SubscriptionName').Trim()
    $subId = ([string]$r.'SubscriptionId').Trim()
    $baRaw = ([string]$r.'BA ID').Trim()

    if (-not $sub -and -not $subId -and -not $baRaw) { continue }   # skip blank rows
    $checked++

    $msg = ""

    if ($subId -notmatch '^[0-9a-fA-F]{8}-([0-9a-fA-F]{4}-){3}[0-9a-fA-F]{12}$') {
        $msg = "INVALID SUBSCRIPTION ID IN FILE ($subId)"
    }
    else {
        $info = Get-SubInfo $subId
        if (-not $info.Found) {
            $msg = "SUBSCRIPTION NOT FOUND / NO ACCESS ($($info.Error))"
        }
        else {
            $tagFound = $false; $tagVal = ""; $similar = ""
            if ($info.Tags) {
                foreach ($p in $info.Tags.PSObject.Properties) {
                    if ($p.Name -ieq $tagName) { $tagFound = $true; $tagVal = [string]$p.Value }
                    elseif ((NormKey $p.Name) -eq (NormKey $tagName)) { $similar = $p.Name + "=" + $p.Value }
                }
            }

            if (-not $tagFound) {
                $msg = "TAG DOES NOT EXIST"
                if ($similar) { $msg += " (similar tag found: $similar)" }
            }
            elseif (-not $tagVal.Trim()) {
                $msg = "TAG EXISTS BUT IS EMPTY"
            }
            elseif ((Norm $tagVal) -ne (Norm $baRaw)) {
                $msg = "MISMATCH - file: $baRaw | Azure tag: $($tagVal.Trim())"
            }
        }
    }

    if ($msg) {
        $line = "[$n/$($rows.Count)] $sub ($subId) -> $msg"
        $problems += $line
        Write-Host $line -ForegroundColor Red
    }
    else {
        $ok++
        Write-Host "[$n/$($rows.Count)] $sub -> OK ($baRaw)" -ForegroundColor Green
    }
}

Write-Host "`n===== SUMMARY =====" -ForegroundColor Cyan
Write-Host "Rows checked : $checked"
Write-Host "Matching     : $ok"
Write-Host "Problems     : $($problems.Count)"

if ($problems.Count -eq 0) {
    Write-Host "`n100% READY" -ForegroundColor Green
}
else {
    Write-Host "`nNOT READY - fix these first:" -ForegroundColor Red
    $problems | ForEach-Object { Write-Host $_ -ForegroundColor Red }
}
