<#
.SYNOPSIS
  Validates the "BA ID" (column C) and "EIM" (column D) values in your
  Excel workbook (path set in -ExcelPath below) against the live tags on each Azure subscription
  (column A = name, column B = subscription ID).

  READ-ONLY and FILE-FREE:
    - Opens the workbook READ-ONLY through Excel (nothing is saved or edited).
    - Reads tags from Azure with Get-AzTag (GET only).
    - Writes NOTHING to disk. All logs and the final summary print on screen.

.PREREQUISITES
  - Microsoft Excel installed on this machine.
  - Az modules:  Install-Module Az.Accounts, Az.Resources -Scope CurrentUser
  - Sign in:     Connect-AzAccount

.USAGE
  1. Set the path on the $ExcelPath line below (or pass -ExcelPath).
  2. .\Verify-SubscriptionTags-NoFiles.ps1
  The workbook may be open or closed in Excel; both work.
#>

[CmdletBinding()]
param(
    # >>> SET YOUR EXCEL FILE PATH HERE <<<
    [string]$ExcelPath    = "C:\Path\To\Stage1Analysis-TagsMapping.xlsx",
    [string]$SheetName    = "Sheet1",
    [int]$MaxRetries      = 3
)

$ErrorActionPreference = "Stop"

function Write-Log {
    param([string]$Message, [string]$Color = "")
    $line = "[{0}] {1}" -f (Get-Date -Format "HH:mm:ss"), $Message
    if ($Color) { Write-Host $line -ForegroundColor $Color } else { Write-Host $line }
}
function Write-Plain {
    param([string]$Message = "", [string]$Color = "")
    if ($Color) { Write-Host $Message -ForegroundColor $Color } else { Write-Host $Message }
}

# ---------- Pre-flight ----------
if (-not (Get-Module -ListAvailable -Name Az.Resources)) {
    throw "Az.Resources module not found. Run: Install-Module Az.Accounts, Az.Resources -Scope CurrentUser"
}
if (-not (Get-AzContext)) { throw "Not signed in to Azure. Run Connect-AzAccount first." }

# ---------- Read the workbook (read-only) ----------
if (-not (Test-Path -LiteralPath $ExcelPath)) {
    throw "Excel file not found: $ExcelPath  (set the correct path in -ExcelPath)"
}
$fullPath = (Resolve-Path -LiteralPath $ExcelPath).Path

$excel = $null; $wb = $null; $ws = $null; $ownsExcel = $false

# If the file is already open in a running Excel, read it from there
$running = $null
try { $running = [Runtime.InteropServices.Marshal]::GetActiveObject("Excel.Application") } catch { $running = $null }
if ($running) {
    foreach ($w in $running.Workbooks) {
        if ($w.FullName -ieq $fullPath) { $wb = $w; $excel = $running; break }
    }
}

try {
    if (-not $wb) {
        # Otherwise open it READ-ONLY in a separate, hidden Excel instance
        $excel = New-Object -ComObject Excel.Application
        $excel.Visible = $false
        $excel.DisplayAlerts = $false
        $ownsExcel = $true
        $wb = $excel.Workbooks.Open($fullPath, 0, $true)    # UpdateLinks=0, ReadOnly=true
    }
    $ws   = $wb.Worksheets.Item($SheetName)
    $data = $ws.UsedRange.Value2                            # one-shot read; changes nothing
}
catch {
    throw "Could not read sheet '$SheetName' from '$fullPath'. If the file is open, make sure no cell is in edit mode. $($_.Exception.Message)"
}
finally {
    if ($ws) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($ws) }
    if ($ownsExcel) {
        # Close WITHOUT saving; only our own hidden instance is quit
        if ($wb)    { try { $wb.Close($false) } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($wb) }
        if ($excel) { try { $excel.Quit() } catch {} ; [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
    }
}
$workbookLabel = Split-Path $fullPath -Leaf

if ($data -isnot [object[,]]) { throw "Sheet '$SheetName' does not contain a data table." }

function ConvertTo-CellString($v) {
    if ($null -eq $v) { return "" }
    if ($v -is [double]) { return $v.ToString("0.##########", [Globalization.CultureInfo]::InvariantCulture) }
    return [string]$v
}

# Locate columns by header name in row 1
$rowCount = $data.GetLength(0)
$colCount = $data.GetLength(1)
$col = @{}
for ($c = 1; $c -le $colCount; $c++) {
    $h = (ConvertTo-CellString $data[1, $c]).Trim()
    if ($h) { $col[$h] = $c }
}
foreach ($h in "SubscriptionName", "SubscriptionId", "BA ID", "EIM") {
    if (-not $col.ContainsKey($h)) {
        throw "Header '$h' not found in row 1. Headers found: $($col.Keys -join ', ')"
    }
}

$rows = for ($r = 2; $r -le $rowCount; $r++) {
    $id = (ConvertTo-CellString $data[$r, $col["SubscriptionId"]]).Trim()
    if (-not $id) { continue }
    [pscustomobject]@{
        Row              = $r
        SubscriptionName = (ConvertTo-CellString $data[$r, $col["SubscriptionName"]]).Trim()
        SubscriptionId   = $id
        BAID             = ConvertTo-CellString $data[$r, $col["BA ID"]]
        EIM              = ConvertTo-CellString $data[$r, $col["EIM"]]
    }
}
$rows = @($rows)

Write-Log "Starting tag verification (read-only, no files written)"
Write-Log "Workbook   : $workbookLabel  |  Sheet: $SheetName"
Write-Log "Azure acct : $((Get-AzContext).Account.Id)"
Write-Log "Subscriptions to check: $($rows.Count)"

$dupes = $rows | Group-Object { $_.SubscriptionId.ToLower() } | Where-Object Count -gt 1
foreach ($d in $dupes) { Write-Log "WARNING: duplicate SubscriptionId in sheet: $($d.Name) ($($d.Count) rows)" "Yellow" }

# ---------- Helpers ----------
function Get-SubscriptionTags {
    param([string]$SubId)
    for ($attempt = 1; $attempt -le $MaxRetries; $attempt++) {
        try {
            $t = Get-AzTag -ResourceId "/subscriptions/$SubId" -ErrorAction Stop
            return $t.Properties.TagsProperty
        }
        catch {
            $msg = $_.Exception.Message
            if ($msg -match "AuthorizationFailed|SubscriptionNotFound|InvalidAuthenticationToken|not found|Forbidden|does not have authorization") { throw }
            if ($attempt -eq $MaxRetries) { throw }
            Start-Sleep -Seconds (2 * $attempt)
        }
    }
}

function Get-TagValue {
    param($Tags, [string]$Name)
    if (-not $Tags) { return @{ Exists = $false; Value = $null } }
    foreach ($k in $Tags.Keys) {
        if ($k -ieq $Name) { return @{ Exists = $true; Value = "$($Tags[$k])" } }
    }
    return @{ Exists = $false; Value = $null }
}

function Compare-TagValue {
    param([string]$Excel, [bool]$AzureExists, [string]$Azure)
    if (-not $AzureExists) {
        if ([string]::IsNullOrEmpty($Excel)) { return "MATCH" }
        return "TAG_MISSING_IN_AZURE"
    }
    if ($Excel -ceq $Azure) { return "MATCH" }
    if ($Excel.Trim() -ceq $Azure.Trim()) { return "WHITESPACE_DIFFERENCE" }
    if ($Excel.Trim() -ieq $Azure.Trim()) { return "CASE_DIFFERENCE" }
    if ([string]::IsNullOrEmpty($Excel))  { return "EMPTY_IN_EXCEL" }
    return "VALUE_MISMATCH"
}

$TagsToCheck = @(
    @{ Tag = "BA ID"; Prop = "BAID" },
    @{ Tag = "EIM";   Prop = "EIM"  }
)

# ---------- Main loop ----------
$results = New-Object System.Collections.Generic.List[object]
$i = 0

foreach ($row in $rows) {
    $i++
    Write-Progress -Activity "Verifying subscription tags" -Status "$i of $($rows.Count): $($row.SubscriptionName)" -PercentComplete (($i / $rows.Count) * 100)

    $issues = @()
    $fetchError = $null
    $tags = $null

    try { $tags = Get-SubscriptionTags -SubId $row.SubscriptionId }
    catch { $fetchError = ($_.Exception.Message -split "`n")[0].Trim() }

    if (-not $fetchError) {
        foreach ($t in $TagsToCheck) {
            $excelVal = $row.($t.Prop)
            $az       = Get-TagValue -Tags $tags -Name $t.Tag
            $status   = Compare-TagValue -Excel $excelVal -AzureExists $az.Exists -Azure $az.Value
            if ($status -ne "MATCH") {
                $issues += [pscustomobject]@{
                    Tag        = $t.Tag
                    Issue      = $status
                    ExcelValue = $excelVal
                    AzureValue = if ($az.Exists) { $az.Value } else { "<tag not set>" }
                }
            }
        }
    }

    $overall = if ($fetchError) { "ERROR" } elseif ($issues.Count -eq 0) { "MATCH" } else { "MISMATCH" }
    $details = if ($fetchError) { $fetchError } else { ($issues | ForEach-Object { "$($_.Tag): $($_.Issue)" }) -join "; " }

    $results.Add([pscustomobject]@{
        Row = $row.Row; SubscriptionName = $row.SubscriptionName; SubscriptionId = $row.SubscriptionId
        Status = $overall; Details = $details; Issues = $issues
    })

    switch ($overall) {
        "MATCH"    { Write-Log "OK        $($row.SubscriptionName) ($($row.SubscriptionId))" }
        "ERROR"    { Write-Log "ERROR     $($row.SubscriptionName) ($($row.SubscriptionId)) - $fetchError" "Red" }
        "MISMATCH" { Write-Log "MISMATCH  $($row.SubscriptionName) ($($row.SubscriptionId)) - $details" "Yellow" }
    }
}
Write-Progress -Activity "Verifying subscription tags" -Completed

# ---------- Final summary (all together) ----------
$total    = $results.Count
$matched  = @($results | Where-Object Status -eq "MATCH").Count
$mismatch = @($results | Where-Object Status -eq "MISMATCH")
$errored  = @($results | Where-Object Status -eq "ERROR")
$pct      = if ($total) { [math]::Round(($matched / $total) * 100, 2) } else { 0 }

Write-Plain ""
Write-Plain ("=" * 78)
Write-Plain "  TAG VERIFICATION SUMMARY   (tags checked: BA ID = col C, EIM = col D)"
Write-Plain ("=" * 78)
Write-Plain "  Run time              : $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Plain "  Subscriptions checked : $total"
Write-Plain "  Matched               : $matched" "Green"
Write-Plain "  Mismatched            : $($mismatch.Count)" $(if ($mismatch.Count) { "Yellow" } else { "Green" })
Write-Plain "  Errors (not verified) : $($errored.Count)"  $(if ($errored.Count)  { "Red" }    else { "Green" })
Write-Plain "  Match rate            : $pct%"
Write-Plain ("-" * 78)

if ($matched -eq $total -and $total -gt 0) {
    Write-Plain "  RESULT: 100% MATCH - every BA ID and EIM value in the sheet equals the live Azure tag." "Green"
}
else {
    Write-Plain "  RESULT: NOT A 100% MATCH - $($mismatch.Count + $errored.Count) subscription(s) need attention." "Red"

    if ($mismatch.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  Issue breakdown:"
        $mismatch | ForEach-Object { $_.Issues } | Group-Object Tag, Issue | Sort-Object Name |
            ForEach-Object { Write-Plain ("    {0,-40} {1}" -f $_.Name, $_.Count) }

        Write-Plain ""
        Write-Plain "  MISMATCHED SUBSCRIPTIONS ($($mismatch.Count)):" "Yellow"
        foreach ($m in $mismatch) {
            Write-Plain ""
            Write-Plain "  * $($m.SubscriptionName)   (Excel row $($m.Row))"
            Write-Plain "    ID: $($m.SubscriptionId)"
            foreach ($iss in $m.Issues) {
                Write-Plain ("    - {0,-6} {1,-22} Excel='{2}'  Azure='{3}'" -f $iss.Tag, $iss.Issue, $iss.ExcelValue, $iss.AzureValue)
            }
        }
    }

    if ($errored.Count -gt 0) {
        Write-Plain ""
        Write-Plain "  COULD NOT BE VERIFIED ($($errored.Count)) - check access / tenant / sign-in:" "Red"
        foreach ($e in $errored) {
            Write-Plain "  * $($e.SubscriptionName)   (Excel row $($e.Row))  [$($e.SubscriptionId)]"
            Write-Plain "    $($e.Details)"
        }
    }
}
Write-Plain ("=" * 78)
Write-Plain "  No files were created or modified. Excel workbook left untouched."
Write-Plain ("=" * 78)
