# Sign in and select the subscription
Connect-AzAccount
$subName = "hsbc-multi-corecomp-nonprod-01"
Set-AzContext -Subscription $subName

# Build the subscription resource ID
$subId = (Get-AzContext).Subscription.Id
$scope = "/subscriptions/$subId"

# New values (replace with yours)
$newTags = @{
    "BA ID"  = "BA0009949"
    "EHC ID" = "RITM32949150"
}

# Merge: updates only these two tags, keeps all others
Update-AzTag -ResourceId $scope -Tag $newTags -Operation Merge

# Verify
(Get-AzTag -ResourceId $scope).Properties.TagsProperty
