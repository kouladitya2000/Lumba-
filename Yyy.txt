$src    = "C:\Path\YourFile.xlsx"      # your workbook
$outDir = "C:\Path\ITSO_Files"         # output folder

New-Item -ItemType Directory -Force -Path $outDir | Out-Null
$excel = New-Object -ComObject Excel.Application
$excel.Visible = $false
$excel.DisplayAlerts = $false

$wb   = $excel.Workbooks.Open($src)
$ws   = $wb.Worksheets.Item("Sheet1")
$last = $ws.Cells.Item($ws.Rows.Count, 1).End(-4162).Row
$data = $ws.Range("A2:G$last").Value2   # A=subName B=ITSO E=EIM F=EHC ID G=BA ID

# Group rows by ITSO (case-insensitive)
$groups = @{}
for ($i = 1; $i -le ($last - 1); $i++) {
    $itso = ("" + $data[$i,2]).Trim()
    if (-not $itso) { continue }
    $key = $itso.ToLower()
    if (-not $groups.ContainsKey($key)) { $groups[$key] = New-Object System.Collections.ArrayList }
    [void]$groups[$key].Add($i)
}

foreach ($key in $groups.Keys) {
    $rows = $groups[$key]
    $n    = $rows.Count
    $arr  = New-Object 'object[,]' ($n + 1), 5
    $arr[0,0] = "ITSO"; $arr[0,1] = "SUBSCRIPTION NAME"
    $arr[0,2] = "EIM";  $arr[0,3] = "EHC ID"; $arr[0,4] = "BA ID"

    for ($r = 0; $r -lt $n; $r++) {
        $i = $rows[$r]
        $arr[($r+1),0] = $key
        $arr[($r+1),1] = $data[$i,1]
        $arr[($r+1),2] = $data[$i,5]
        $arr[($r+1),3] = $data[$i,6]
        $arr[($r+1),4] = $data[$i,7]
    }

    $nb = $excel.Workbooks.Add()
    $s  = $nb.Worksheets.Item(1)
    $s.Range($s.Cells.Item(1,1), $s.Cells.Item($n+1,5)).Value2 = $arr
    $s.Rows.Item(1).Font.Bold = $true
    $s.Columns.AutoFit() | Out-Null

    $safe = ($key -replace '[\\/:*?"<>|]', '_')
    $nb.SaveAs("$outDir\$safe.xlsx", 51)
    $nb.Close($false)
}

$wb.Close($false)
$excel.Quit()
Write-Host "Done. Created $($groups.Count) files in $outDir"
