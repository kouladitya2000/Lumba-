# ================= Settings (edit these) =================
$inputCsv  = "$env:USERPROFILE\Downloads\ValidatedBAFINAL.csv"
$outputCsv = "$env:USERPROFILE\Downloads\SingleFinalEHC_" + (Get-Date -Format "yyyyMMdd_HHmmss") + ".csv"
# =========================================================

if (-not (Test-Path $inputCsv)) { throw "Input file not found: $inputCsv" }
if (-not (Test-Path (Split-Path $outputCsv))) { throw "Output folder not found: $(Split-Path $outputCsv)" }

$rows = @(Import-Csv $inputCsv -Encoding UTF8)
if ($rows.Count -eq 0) { throw "Input CSV loaded 0 rows" }
$cols = $rows[0].PSObject.Properties.Name
foreach ($need in 'SubscriptionName','SubscriptionId','BA ID','EHC ID') {
    if ($cols -notcontains $need) { throw "Column '$need' not found. Headers: $($cols -join ', ')" }
}

function Q($v) { '"' + ([string]$v -replace '"', '""') + '"' }

$lines = @()
$lines += (Q 'SubscriptionName') + "," + (Q 'SubscriptionId') + "," + (Q 'BA ID') + "," + (Q 'EHC ID')

$cntSingle = 0; $cntMulti = 0; $cntKept = 0

foreach ($r in $rows) {
    $sub    = ([string]$r.'SubscriptionName').Trim()
    $subId  = ([string]$r.'SubscriptionId').Trim()
    $baRaw  = ([string]$r.'BA ID').Trim()
    $ehcRaw = ([string]$r.'EHC ID').Trim()

    # Take the first RITM id (keeps a "(Withdrawn)" label if it has one)
    if ($ehcRaw -match '^(RITM\d+)(\([^)]*\))?') {
        $first = $matches[1] + $matches[2]
        if ($ehcRaw -like '*,*') { $cntMulti++ } else { $cntSingle++ }
    }
    else {
        # NO EHC FOUND / NO WORKING EHC / API NOT WORKING etc: keep as-is
        $first = $ehcRaw
        $cntKept++
    }

    $lines += (Q $sub) + "," + (Q $subId) + "," + (Q $baRaw) + "," + (Q $first)
}

$lines | Out-File -FilePath $outputCsv -Encoding UTF8

Write-Host "`n===== SUMMARY =====" -ForegroundColor Cyan
Write-Host "Rows in input                 : $($rows.Count)"
Write-Host "Rows reduced to first EHC     : $cntMulti"
Write-Host "Rows that already had one EHC : $cntSingle"
Write-Host "Rows kept as-is (no RITM)     : $cntKept"
Write-Host "Output file                   : $outputCsv"
