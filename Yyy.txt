# ============ SETTINGS ============
$baseUrl = "https://cstmg-fa-euw-01.cstmg-ase-euw-01.appserviceenvironment.net"

#>>> PASTE 1: your function key (keep the "code=" part)
$faCode  = "code=<YOUR_FUNCTION_KEY>"

#>>> PASTE 2: full path of your input Excel (the one in your screenshot)
$inputPath  = "$env:USERPROFILE\Desktop\TagsValidator\Stage2SubscriptionTagsReportEHC.xlsx"

#>>> PASTE 3: where the new Excel should be saved (change the name if you like)
$outputPath = "$env:USERPROFILE\Desktop\TagsValidator\BA_EHC_API_Report.xlsx"

$maxAttempts        = 4       # retries for throttling or temporary errors only
$pauseBetween       = 500     # ms pause between calls (raise to 1000-2000 if blocked)
$stopAtFirstWorking = $false  # $true = stop at the first working EHC per row (faster)

# Input columns (L = 12, M = 13) and the columns to keep (A-F, N-P)
$ehcCol   = 12
$baCol    = 13
$keepCols = 1,2,3,4,5,6,14,15,16

# ============ API TEST (do not edit) ============
# 404 / "not found" body = real fail (no retry). Other errors (429, 5xx, timeouts) are retried.
function Test-Pair($ba, $ritm) {
    $url  = "$baseUrl/api/GetBAData?baid=$ba&ehcId=$ritm&$faCode"
    $code = 0
    for ($i = 1; $i -le $maxAttempts; $i++) {
        try {
            $resp = Invoke-WebRequest -UseBasicParsing -Uri $url -Method GET -ContentType "application/json"
            $code = $resp.StatusCode
            if ($resp.StatusCode -eq 200 -and ([string]$resp.Content) -notmatch 'not found') {
                return @{ OK = $true; Code = 200 }
            }
            return @{ OK = $false; Code = $code }
        }
        catch {
            $code = $_.Exception.Response.StatusCode.value__
            if ($code -eq 404) { return @{ OK = $false; Code = 404 } }
        }
        if ($i -lt $maxAttempts) { Start-Sleep -Seconds (3 * $i) }
    }
    return @{ OK = $false; Code = $code }
}

# ============ OPEN INPUT EXCEL (read-only) ============
$excel = New-Object -ComObject Excel.Application
$excel.Visible = $false
$excel.DisplayAlerts = $false

$wbIn    = $excel.Workbooks.Open($inputPath, 0, $true)
$wsIn    = $wbIn.Worksheets.Item(1)
$lastRow = $wsIn.Cells.Item($wsIn.Rows.Count, 1).End(-4162).Row   # last filled row in column A

# ============ CREATE OUTPUT EXCEL + HEADERS ============
$wbOut = $excel.Workbooks.Add()
$wsOut = $wbOut.Worksheets.Item(1)

$c = 1
foreach ($k in $keepCols) {
    $wsOut.Cells.Item(1, $c).Value2 = "$($wsIn.Cells.Item(1, $k).Value2)"
    $c++
}
$wsOut.Cells.Item(1, $c).Value2     = "Onboarding BA ID"
$wsOut.Cells.Item(1, $c + 1).Value2 = "Status Code"
$wsOut.Cells.Item(1, $c + 2).Value2 = "API Works"
$wsOut.Cells.Item(1, $c + 3).Value2 = "Working EHC"
$lastOutCol = $c + 3
$wsOut.Range($wsOut.Cells.Item(1, 1), $wsOut.Cells.Item(1, $lastOutCol)).Font.Bold = $true

# ============ PROCESS EVERY ROW ============
$cache = @{}
$out   = @()
$outRow = 2

for ($r = 2; $r -le $lastRow; $r++) {
    $ehcText = "$($wsIn.Cells.Item($r, $ehcCol).Value2)"
    $baText  = "$($wsIn.Cells.Item($r, $baCol).Value2)"

    $ba = ""
    if ($baText -match 'BA\d+') { $ba = $matches[0] }

    $ritms = @()
    $found = Select-String -InputObject $ehcText -Pattern 'RITM\d+' -AllMatches
    if ($found) { $ritms = @($found.Matches.Value) }

    $working = @()
    $codes   = @()
    if ($ba -and $ritms.Count -gt 0) {
        foreach ($ritm in $ritms) {
            $key = "$ba|$ritm"
            if (-not $cache.ContainsKey($key)) {
                $cache[$key] = Test-Pair $ba $ritm
                Start-Sleep -Milliseconds $pauseBetween
            }
            $codes += "$($cache[$key].Code)"
            if ($cache[$key].OK) {
                $working += $ritm
                if ($stopAtFirstWorking) { break }
            }
        }
    }

    if ($working.Count -gt 0)      { $works = "Yes"; $status = "200"; $workTxt = ($working -join ",") }
    elseif ($codes.Count -gt 0)    { $works = "No";  $status = (($codes | Select-Object -Unique) -join ","); $workTxt = "" }
    else                           { $works = "No";  $status = "N/A"; $workTxt = "" }

    # write the preserved columns + the new columns
    $c = 1
    foreach ($k in $keepCols) {
        $wsOut.Cells.Item($outRow, $c).Value2 = "$($wsIn.Cells.Item($r, $k).Value2)"
        $c++
    }
    $wsOut.Cells.Item($outRow, $c).Value2     = $ba
    $wsOut.Cells.Item($outRow, $c + 1).Value2 = $status
    $wsOut.Cells.Item($outRow, $c + 2).Value2 = $works
    $wsOut.Cells.Item($outRow, $c + 3).Value2 = $workTxt

    # green = works, red = does not work
    $color = 13551615
    if ($works -eq "Yes") { $color = 13561798 }
    $wsOut.Range($wsOut.Cells.Item($outRow, 1), $wsOut.Cells.Item($outRow, $lastOutCol)).Interior.Color = $color

    Write-Host "Row $r  $ba  ->  $works  [$status]  $workTxt"
    $out += New-Object PSObject -Property @{ Row = $r; BAID = $ba; StatusCode = $status; Works = $works; WorkingEHC = $workTxt }
    $outRow++
}

# ============ PRINT IN POWERSHELL ============
$out | Select-Object Row, BAID, StatusCode, Works, WorkingEHC | Format-Table -AutoSize
$yes = @($out | Where-Object { $_.Works -eq "Yes" }).Count
Write-Host "Works: $yes   Not working: $($out.Count - $yes)   Total rows: $($out.Count)"

# ============ SAVE EXCEL ============
$wsOut.Columns.AutoFit() | Out-Null
$wbIn.Close($false)
$wbOut.SaveAs($outputPath, 51)

if (Test-Path $outputPath) {
    Write-Host "Saved: $outputPath"
    $wbOut.Close($false)
    $excel.Quit()
}
else {
    # Same silent-save problem as before: sensitivity label. Keep Excel open so nothing is lost.
    $csvPath = $outputPath -replace '\.xlsx$', '.csv'
    $out | Select-Object Row, BAID, StatusCode, Works, WorkingEHC | Export-Csv $csvPath -NoTypeInformation
    $excel.Visible = $true
    Write-Host "Excel did not save. In the open Excel window set Sensitivity to Internal and press Ctrl+S." -ForegroundColor Yellow
    Write-Host "A CSV backup was saved: $csvPath"
}
